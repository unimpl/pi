  59 export const stream: StreamFunction<"google-generative-ai", GoogleOptions> = (
  60 	model: Model<"google-generative-ai">,
  61 	context: TranscriptContext,
  62 	options?: GoogleOptions,
  63 ): AssistantMessageEventStream => {
  64 	const stream = new AssistantMessageEventStream();
  65 	const normalizedContext = collapseSystemMessages(context);
  66 
  67 	(async () => {
  68 		const output: AssistantMessage = {
  69 			role: "assistant",
  70 			content: [],
  71 			api: "google-generative-ai" as Api,
  72 			provider: model.provider,
  73 			model: model.id,
  74 			usage: {
  75 				input: 0,
  76 				output: 0,
  77 				cacheRead: 0,
  78 				cacheWrite: 0,
  79 				totalTokens: 0,
  80 				cost: { input: 0, output: 0, cacheRead: 0, cacheWrite: 0, total: 0 },
  81 			},
  82 			stopReason: "pending",
  83 			timestamp: Date.now(),
  84 		};
  85 
  86 		try {
  87 			if (options?.fetch && options.fetch !== globalThis.fetch) {
  88 				throw new Error("Custom fetch is not supported by the Google Generative AI adapter");
  89 			}
  90 			const apiKey = options?.apiKey;
  91 			if (!apiKey) {
  92 				throw new Error(`No API key for provider: ${model.provider}`);
  93 			}
  94 			const client = createClient(model, apiKey, options?.headers);
  95 			let params = buildParams(model, normalizedContext, options);
  96 			const nextParams = await options?.onPayload?.(params, model);
  97 			if (nextParams !== undefined) {
  98 				params = nextParams as GenerateContentParameters;
  99 			}
 100 			const googleStream = await retryGoogleRequest(() => client.models.generateContentStream(params), options);
 101 
 102 			stream.push({ type: "start", partial: output });
 103 			let currentBlock: TextContent | ThinkingContent | null = null;
 104 			const blocks = output.content;
 105 			const blockIndex = () => blocks.length - 1;
 106 			for await (const chunk of googleStream) {
 107 				// @google/genai documents GenerateContentResponse.responseId as an output-only field
 108 				// used to identify each response. Keep the first non-empty one from the stream.
 109 				output.responseId ||= chunk.responseId;
 110 				const candidate = chunk.candidates?.[0];
 111 				if (candidate?.content?.parts) {
 112 					for (const part of candidate.content.parts) {
 113 						if (part.text !== undefined) {
 114 							const isThinking = isThinkingPart(part);
 115 							if (
 116 								!currentBlock ||
 117 								(isThinking && currentBlock.type !== "thinking") ||
 118 								(!isThinking && currentBlock.type !== "text")
 119 							) {
 120 								if (currentBlock) {
 121 									if (currentBlock.type === "text") {
 122 										stream.push({
 123 											type: "text_end",
 124 											contentIndex: blocks.length - 1,
 125 											content: currentBlock.text,
 126 											partial: output,
 127 										});
 128 									} else {
 129 										stream.push({
 130 											type: "thinking_end",
 131 											contentIndex: blockIndex(),
 132 											content: currentBlock.thinking,
 133 											partial: output,
 134 										});
 135 									}
 136 								}
 137 								if (isThinking) {
 138 									currentBlock = { type: "thinking", thinking: "", thinkingSignature: undefined };
 139 									output.content.push(currentBlock);
 140 									stream.push({ type: "thinking_start", contentIndex: blockIndex(), partial: output });
 141 								} else {
 142 									currentBlock = { type: "text", text: "" };
 143 									output.content.push(currentBlock);
 144 									stream.push({ type: "text_start", contentIndex: blockIndex(), partial: output });
 145 								}
 146 							}
 147 							if (currentBlock.type === "thinking") {
 148 								currentBlock.thinking += part.text;
 149 								currentBlock.thinkingSignature = retainThoughtSignature(
 150 									currentBlock.thinkingSignature,
 151 									part.thoughtSignature,
 152 								);
 153 								stream.push({
 154 									type: "thinking_delta",
 155 									contentIndex: blockIndex(),
 156 									delta: part.text,
 157 									partial: output,
 158 								});
 159 							} else {
 160 								currentBlock.text += part.text;
 161 								currentBlock.textSignature = retainThoughtSignature(
 162 									currentBlock.textSignature,
 163 									part.thoughtSignature,
 164 								);
 165 								stream.push({
 166 									type: "text_delta",
 167 									contentIndex: blockIndex(),
 168 									delta: part.text,
 169 									partial: output,
 170 								});
 171 							}
 172 						}
 173 
 174 						if (part.functionCall) {
 175 							if (currentBlock) {
 176 								if (currentBlock.type === "text") {
 177 									stream.push({
 178 										type: "text_end",
 179 										contentIndex: blockIndex(),
 180 										content: currentBlock.text,
 181 										partial: output,
 182 									});
 183 								} else {
 184 									stream.push({
 185 										type: "thinking_end",
 186 										contentIndex: blockIndex(),
 187 										content: currentBlock.thinking,
 188 										partial: output,
 189 									});
 190 								}
 191 								currentBlock = null;
 192 							}
 193 
 194 							// Generate unique ID if not provided or if it's a duplicate
 195 							const providedId = part.functionCall.id;
 196 							const needsNewId =
 197 								!providedId || output.content.some((b) => b.type === "toolCall" && b.id === providedId);
 198 							const toolCallId = needsNewId
 199 								? `${part.functionCall.name}_${Date.now()}_${++toolCallCounter}`
 200 								: providedId;
 201 
 202 							const toolCall: ToolCall = {
 203 								type: "toolCall",
 204 								id: toolCallId,
 205 								name: part.functionCall.name || "",
 206 								arguments: (part.functionCall.args as Record<string, any>) ?? {},
 207 								...(part.thoughtSignature && { thoughtSignature: part.thoughtSignature }),
 208 							};
 209 
 210 							output.content.push(toolCall);
 211 							stream.push({ type: "toolcall_start", contentIndex: blockIndex(), partial: output });
 212 							stream.push({
 213 								type: "toolcall_delta",
 214 								contentIndex: blockIndex(),
 215 								delta: JSON.stringify(toolCall.arguments),
 216 								partial: output,
 217 							});
 218 							stream.push({ type: "toolcall_end", contentIndex: blockIndex(), toolCall, partial: output });
 219 						}
 220 					}
 221 				}
 222 
 223 				if (candidate?.finishReason) {
 224 					output.rawStopReason = candidate.finishReason;
 225 					output.stopReason = mapStopReason(candidate.finishReason);
 226 					if (output.content.some((b) => b.type === "toolCall") && output.stopReason === "stop") {
 227 						output.stopReason = "toolUse";
 228 					}
 229 				}
 230 
 231 				if (chunk.usageMetadata) {
 232 					output.usage = {
 233 						input:
 234 							(chunk.usageMetadata.promptTokenCount || 0) - (chunk.usageMetadata.cachedContentTokenCount || 0),
 235 						output:
 236 							(chunk.usageMetadata.candidatesTokenCount || 0) + (chunk.usageMetadata.thoughtsTokenCount || 0),
 237 						cacheRead: chunk.usageMetadata.cachedContentTokenCount || 0,
 238 						cacheWrite: 0,
 239 						reasoning: chunk.usageMetadata.thoughtsTokenCount || 0,
 240 						totalTokens: chunk.usageMetadata.totalTokenCount || 0,
 241 						cost: {
 242 							input: 0,
 243 							output: 0,
 244 							cacheRead: 0,
 245 							cacheWrite: 0,
 246 							total: 0,
 247 						},
 248 					};
 249 					calculateCost(model, output.usage);
 250 				}
 251 			}
 252 
 253 			if (currentBlock) {
 254 				if (currentBlock.type === "text") {
 255 					stream.push({
 256 						type: "text_end",
 257 						contentIndex: blockIndex(),
 258 						content: currentBlock.text,
 259 						partial: output,
 260 					});
 261 				} else {
 262 					stream.push({
 263 						type: "thinking_end",
 264 						contentIndex: blockIndex(),
 265 						content: currentBlock.thinking,
 266 						partial: output,
 267 					});
 268 				}
 269 			}
 270 
 271 			if (options?.signal?.aborted) {
 272 				throw new Error("Request was aborted");
 273 			}
 274 
 275 			if (output.stopReason === "pending") {
 276 				throw new Error("Google stream ended without a finish reason");
 277 			}
 278 			if (output.stopReason === "aborted" || output.stopReason === "error") {
 279 				const errorMessage = output.rawStopReason
 280 					? `Provider stopped with: ${output.rawStopReason}`
 281 					: "An unknown error occurred";
 282 				throw new Error(errorMessage);
 283 			}
 284 
 285 			stream.push({ type: "done", reason: output.stopReason, message: output });
 286 			stream.end();
 287 		} catch (error) {
 288 			// Remove internal index property used during streaming
 289 			for (const block of output.content) {
 290 				if ("index" in block) {
 291 					delete (block as { index?: number }).index;
 292 				}
 293 			}
 294 			output.stopReason = options?.signal?.aborted ? "aborted" : "error";
 295 			output.errorMessage = formatProviderError(normalizeProviderError(error));
 296 			stream.push({ type: "error", reason: output.stopReason, error: output });
 297 			stream.end();
 298 		}
 299 	})();
 300 
 301 	return stream;
 302 };

 304 export const streamSimple: StreamFunction<"google-generative-ai", SimpleStreamOptions> = (
 305 	model: Model<"google-generative-ai">,
 306 	context: TranscriptContext,
 307 	options?: SimpleStreamOptions,
 308 ): AssistantMessageEventStream => {
 309 	const apiKey = options?.apiKey;
 310 	if (!apiKey) {
 311 		throw new Error(`No API key for provider: ${model.provider}`);
 312 	}
 313 
 314 	const base = {
 315 		...buildBaseOptions(model, context, options, apiKey),
 316 		toolChoice: options?.toolChoice,
 317 	} satisfies GoogleOptions;
 318 	if (!options?.reasoning) {
 319 		return stream(model, context, { ...base, thinking: { enabled: false } } satisfies GoogleOptions);
 320 	}
 321 
 322 	const clampedReasoning = clampThinkingLevel(model, options.reasoning);
 323 	if (clampedReasoning === "off") {
 324 		return stream(model, context, { ...base, thinking: { enabled: false } } satisfies GoogleOptions);
 325 	}
 326 	const resolvedLevel = resolveGoogleThinkingLevel(model, clampedReasoning);
 327 
 328 	if (usesGoogleThinkingLevel(model)) {
 329 		return stream(model, context, {
 330 			...base,
 331 			thinking: {
 332 				enabled: true,
 333 				level: toGoogleThinkingLevel(resolvedLevel),
 334 			},
 335 		} satisfies GoogleOptions);
 336 	}
 337 
 338 	return stream(model, context, {
 339 		...base,
 340 		thinking: {
 341 			enabled: true,
 342 			budgetTokens: getGoogleBudget(model, resolvedLevel, options.thinkingBudgets),
 343 		},
 344 	} satisfies GoogleOptions);
 345 };

 347 function createClient(
 348 	model: Model<"google-generative-ai">,
 349 	apiKey?: string,
 350 	optionsHeaders?: ProviderHeaders,
 351 ): GoogleGenAI {
 352 	const httpOptions: { baseUrl?: string; apiVersion?: string; headers?: Record<string, string> } = {};
 353 	if (model.baseUrl) {
 354 		httpOptions.baseUrl = model.baseUrl;
 355 		httpOptions.apiVersion = ""; // baseUrl already includes version path, don't append
 356 	}
 357 	const headers = providerHeadersToRecord({ "User-Agent": getPiUserAgent(), ...model.headers, ...optionsHeaders });
 358 	if (headers) {
 359 		httpOptions.headers = headers;
 360 	}
 361 
 362 	return new GoogleGenAI({
 363 		apiKey,
 364 		httpOptions: Object.keys(httpOptions).length > 0 ? httpOptions : undefined,
 365 	});
 366 }

 368 function buildParams(
 369 	model: Model<"google-generative-ai">,
 370 	context: TranscriptContext,
 371 	options: GoogleOptions = {},
 372 ): GenerateContentParameters {
 373 	const contents = convertMessages(model, context);
 374 	const initialSystemMessage = getInitialSystemMessage(context.messages);
 375 	const currentTools = getCurrentTools(context.messages);
 376 
 377 	const generationConfig: GenerateContentConfig = {};
 378 	if (options.temperature !== undefined) {
 379 		generationConfig.temperature = options.temperature;
 380 	}
 381 	if (options.maxTokens !== undefined) {
 382 		generationConfig.maxOutputTokens = options.maxTokens;
 383 	}
 384 
 385 	const supportsStrictMode = supportsGoogleStrictToolSampling(model.id);
 386 	const functionCallingMode =
 387 		currentTools.length > 0
 388 			? resolveGoogleFunctionCallingMode(currentTools, options.toolChoice, supportsStrictMode)
 389 			: undefined;
 390 	const systemInstruction = initialSystemMessage ? getSystemMessageText(initialSystemMessage) : "";
 391 	const config: GenerateContentConfig = {
 392 		...(Object.keys(generationConfig).length > 0 && generationConfig),
 393 		...(systemInstruction && { systemInstruction: sanitizeSurrogates(systemInstruction) }),
 394 		...(currentTools.length > 0 && {
 395 			tools: convertTools(currentTools, false, supportsStrictMode),
 396 		}),
 397 		...(functionCallingMode !== undefined && {
 398 			toolConfig: { functionCallingConfig: { mode: functionCallingMode } },
 399 		}),
 400 	};
 401 
 402 	if (options.thinking?.enabled && model.reasoning) {
 403 		const thinkingConfig: ThinkingConfig = { includeThoughts: true };
 404 		if (options.thinking.level !== undefined) {
 405 			thinkingConfig.thinkingLevel = toGoogleSdkThinkingLevel(options.thinking.level);
 406 		} else if (options.thinking.budgetTokens !== undefined) {
 407 			thinkingConfig.thinkingBudget = options.thinking.budgetTokens;
 408 		}
 409 		config.thinkingConfig = thinkingConfig;
 410 	} else if (model.reasoning && options.thinking && !options.thinking.enabled) {
 411 		config.thinkingConfig = getDisabledGoogleThinkingConfig(model);
 412 	}
 413 
 414 	if (options.signal) {
 415 		if (options.signal.aborted) {
 416 			throw new Error("Request aborted");
 417 		}
 418 		config.abortSignal = options.signal;
 419 	}
 420 
 421 	const params: GenerateContentParameters = {
 422 		model: model.id,
 423 		contents,
 424 		config,
 425 	};
 426 
 427 	return params;
 428 }

 430 function getGoogleBudget(
 431 	model: Model<"google-generative-ai">,
 432 	level: ResolvedGoogleThinkingLevel,
 433 	customBudgets?: ThinkingBudgets,
 434 ): number {
 435 	if (customBudgets?.[level] !== undefined) {
 436 		return customBudgets[level]!;
 437 	}
 438 
 439 	if (model.id.includes("2.5-pro")) {
 440 		const budgets: Record<ResolvedGoogleThinkingLevel, number> = {
 441 			minimal: 128,
 442 			low: 2048,
 443 			medium: 8192,
 444 			high: 32768,
 445 		};
 446 		return budgets[level];
 447 	}
 448 
 449 	if (model.id.includes("2.5-flash-lite")) {
 450 		const budgets: Record<ResolvedGoogleThinkingLevel, number> = {
 451 			minimal: 512,
 452 			low: 2048,
 453 			medium: 8192,
 454 			high: 24576,
 455 		};
 456 		return budgets[level];
 457 	}
 458 
 459 	if (model.id.includes("2.5-flash")) {
 460 		const budgets: Record<ResolvedGoogleThinkingLevel, number> = {
 461 			minimal: 128,
 462 			low: 2048,
 463 			medium: 8192,
 464 			high: 24576,
 465 		};
 466 		return budgets[level];
 467 	}
 468 
 469 	return -1;
