 135 /**
 136  * Parse a skill block from message text.
 137  * Returns null if the text doesn't contain a skill block.
 138  */
 139 export function parseSkillBlock(text: string): ParsedSkillBlock | null {
 140 	const match = text.match(/^<skill name="([^"]+)" location="([^"]+)">\n([\s\S]*?)\n<\/skill>(?:\n\n([\s\S]+))?$/);
 141 	if (!match) return null;
 142 	return {
 143 		name: match[1],
 144 		location: match[2],
 145 		content: match[3],
 146 		userMessage: match[4]?.trim() || undefined,
 147 	};

 420 	private async _getRequiredRequestAuth(model: Model<any>): Promise<{
 421 		model: Model<any>;
 422 		apiKey?: string;
 423 		headers?: Record<string, string>;
 424 		env?: Record<string, string>;
 425 	}> {
 426 		let result: AuthResult | undefined;
 427 		try {
 428 			result = await this._modelRuntime.getAuth(model);
 429 		} catch (error) {
 430 			const cause = error instanceof Error ? error.cause : undefined;
 431 			if (cause instanceof Error && cause.message === "authHeader requires a resolved API key") {
 432 				throw new Error(formatNoApiKeyFoundMessage(model.provider));
 433 			}
 434 			throw error;
 435 		}
 436 		if (result && (result.auth.apiKey || result.auth.headers)) {
 437 			const requestModel = result.auth.baseUrl ? { ...model, baseUrl: result.auth.baseUrl } : model;
 438 			return {
 439 				model: requestModel,
 440 				apiKey: result.auth.apiKey,
 441 				headers: withoutDeletedHeaders(result.auth.headers),
 442 				env: result.env,
 443 			};
 444 		}
 445 
 446 		const isOAuth = this._modelRuntime.isUsingOAuth(model.provider);
 447 		if (isOAuth) {
 448 			throw new Error(
 449 				`Authentication failed for "${model.provider}". ` +
 450 					`Credentials may have expired or network is unavailable. ` +
 451 					`Run '/login ${model.provider}' to re-authenticate.`,
 452 			);
 453 		}
 454 		throw new Error(formatNoApiKeyFoundMessage(model.provider));
 455 	}

 482 	/**
 483 	 * Install tool hooks once on the Agent instance.
 484 	 *
 485 	 * The callbacks read `this._extensionRunner` at execution time, so extension reload swaps in the
 486 	 * new runner without reinstalling hooks. Extension-specific tool wrappers are still used to adapt
 487 	 * registered tool execution to the extension context. Tool call and tool result interception now
 488 	 * happens here instead of in wrappers.
 489 	 */
 490 	private _installAgentToolHooks(): void {
 491 		this.agent.beforeToolCall = async ({ toolCall, args }) => {
 492 			const runner = this._extensionRunner;
 493 			if (!runner.hasHandlers("tool_call")) {
 494 				return undefined;
 495 			}
 496 
 497 			try {
 498 				return await runner.emitToolCall({
 499 					type: "tool_call",
 500 					toolName: toolCall.name,
 501 					toolCallId: toolCall.id,
 502 					input: args as Record<string, unknown>,
 503 				});
 504 			} catch (err) {
 505 				if (err instanceof Error) {
 506 					throw err;
 507 				}
 508 				throw new Error(`Extension failed, blocking execution: ${String(err)}`);
 509 			}
 510 		};
 511 
 512 		this.agent.afterToolCall = async ({ toolCall, args, result, isError }) => {
 513 			const runner = this._extensionRunner;
 514 			const hookResult = runner.hasHandlers("tool_result")
 515 				? await runner.emitToolResult({
 516 						type: "tool_result",
 517 						toolName: toolCall.name,
 518 						toolCallId: toolCall.id,
 519 						input: args as Record<string, unknown>,
 520 						content: result.content,
 521 						details: result.details,
 522 						isError,
 523 						usage: result.usage,
 524 					})
 525 				: undefined;
 526 
 527 			const content = hookResult?.content ?? result.content ?? [];
 528 			// Runs after the extension hook so images injected or replaced by extensions are normalized too.
 529 			const normalizedContent = await normalizeToolResultImages(content, {
 530 				autoResizeImages: this.settingsManager.getImageAutoResize(),
 531 			});
 532 
 533 			if (!hookResult && normalizedContent === content) {
 534 				return undefined;
 535 			}
 536 
 537 			return {
 538 				content: normalizedContent,
 539 				details: hookResult?.details,
 540 				isError: hookResult?.isError ?? isError,
 541 				usage: hookResult?.usage,
 542 			};
 543 		};
 544 	}

 565 	private _installAgentNextTurnRefresh(): void {
 566 		const previousPrepareNextTurnWithContext =
 567 			this.agent.prepareNextTurnWithContext ??

 573 			const previousSnapshot = await previousPrepareNextTurnWithContext?.({ ...turn, context }, signal);

 575 			const runOptions = this._runSystemPromptOptions ?? this._baseSystemPromptOptions;

 583 			// Keep session.systemPrompt and ctx.getSystemPrompt() in step with what the provider sees.
 584 			this._runSystemPromptOptions = options;

 658 	/** Internal handler for agent events - shared by subscribe and reconnect */
 659 	private _handleAgentEvent = async (event: AgentEvent): Promise<void> => {
 660 		// When a user message starts, check if it's from either queue and remove it BEFORE emitting
 661 		// This ensures the UI sees the updated queue state
 662 		if (event.type === "message_start" && event.message.role === "user") {
 663 			this._overflowRecoveryAttempted = false;
 664 			const messageText = contentText(event.message.content, "");
 665 			if (messageText) {
 666 				// Check steering queue first
 667 				const steeringIndex = this._steeringMessages.indexOf(messageText);
 668 				if (steeringIndex !== -1) {
 669 					this._steeringMessages.splice(steeringIndex, 1);
 670 					this._emitQueueUpdate();
 671 				} else {
 672 					// Check follow-up queue
 673 					const followUpIndex = this._followUpMessages.indexOf(messageText);
 674 					if (followUpIndex !== -1) {
 675 						this._followUpMessages.splice(followUpIndex, 1);
 676 						this._emitQueueUpdate();
 677 					}
 678 				}
 679 			}
 680 		}
 681 
 682 		// Emit to extensions first
 683 		await this._emitExtensionEvent(event);
 684 
 685 		// Notify all listeners
 686 		this._emit(event.type === "agent_end" ? { ...event, willRetry: this._willRetryAfterAgentEnd(event) } : event);
 687 
 688 		// Handle session persistence
 689 		if (event.type === "message_end") {
 690 			// Check if this is a custom message from extensions
 691 			if (event.message.role === "custom") {
 692 				// Persist as CustomMessageEntry
 693 				this.sessionManager.appendCustomMessageEntry(
 694 					event.message.customType,
 695 					event.message.content,
 696 					event.message.display,
 697 					event.message.details,
 698 				);
 699 			} else if (
 700 				event.message.role === "system" ||
 701 				event.message.role === "user" ||
 702 				event.message.role === "assistant" ||
 703 				event.message.role === "toolResult"
 704 			) {
 705 				// Regular LLM message - persist as SessionMessageEntry
 706 				this.sessionManager.appendMessage(event.message);
 707 			}
 708 			// Other message types (bashExecution, compactionSummary, branchSummary) are persisted elsewhere
 709 
 710 			// Track assistant message for auto-compaction (checked on agent_end)
 711 			if (event.message.role === "assistant") {
 712 				this._lastAssistantMessage = event.message;
 713 
 714 				const assistantMsg = event.message as AssistantMessage;
 715 				if (assistantMsg.stopReason !== "error" && assistantMsg.stopReason !== "length") {
 716 					this._overflowRecoveryAttempted = false;
 717 				}
 718 
 719 				// Reset retry counter immediately on successful assistant response
 720 				// This prevents accumulation across multiple LLM calls within a turn
 721 				if (assistantMsg.stopReason !== "error" && this._retryAttempt > 0) {
 722 					this._emit({
 723 						type: "auto_retry_end",
 724 						success: true,
 725 						attempt: this._retryAttempt,
 726 					});
 727 					this._retryAttempt = 0;
 728 				}
 729 			}
 730 		}
 731 
 732 		// A turn ends after its assistant message and every tool result has been appended,
 733 		// so this is the first point in the run where a context-only custom message can be
 734 		// inserted without landing between a tool call and its result. Flushing after the
 735 		// extension and listener dispatch above also picks up messages that turn_end
 736 		// handlers queued.
 737 		if (event.type === "turn_end") {
 738 			this._flushPendingCustomMessages();
 739 		}
 740 	};

 742 	private _willRetryAfterAgentEnd(event: Extract<AgentEvent, { type: "agent_end" }>): boolean {
 743 		const settings = this.settingsManager.getRetrySettings();
 744 		if (!settings.enabled || this._retryAttempt >= settings.maxRetries) {
 745 			return false;

 747

 749 			const message = event.messages[i];
 750 			if (message.role === "assistant") {
 751 				return this._isRetryableError(message as AssistantMessage);
 752 			}
 753 		}
 754 		return false;
 755 	}

 769 	private _replaceMessageInPlace(target: AgentMessage, replacement: AgentMessage): void {
 770 		// Agent-core stores the finalized message object in its state before emitting message_end.
 771 		// SessionManager persistence happens later in _handleAgentEvent() with event.message.
 772 		// Mutating this object in place keeps agent state, later turn/agent events, listeners,
 773 		// and the eventual SessionManager.appendMessage(event.message) persistence in sync.
 774 		if (target === replacement) {
 775 			return;
 776 		}
 777 
 778 		const targetRecord = target as unknown as Record<string, unknown>;
 779 		for (const key of Object.keys(targetRecord)) {
 780 			delete targetRecord[key];
 781 		}
 782 		Object.assign(targetRecord, replacement);
 783 	}

1191 	private async _handlePostAgentRun(): Promise<boolean> {
1192 		const msg = this._lastAssistantMessage;
1193 		this._lastAssistantMessage = undefined;
1194 		if (!msg) {
1195 			return false;
1196 		}
1197 
1198 		if (this._isRetryableError(msg) && (await this._prepareRetry(msg))) {
1199 			return true;
1200 		}
1201 
1202 		if (msg.stopReason === "error" && this._retryAttempt > 0) {
1203 			this._emit({
1204 				type: "auto_retry_end",
1205 				success: false,
1206 				attempt: this._retryAttempt,
1207 				finalError: msg.errorMessage,
1208 			});

1243 	 * - Handles extension commands (registered via pi.registerCommand) immediately, even during streaming
1244 	 * - Expands file-based prompt templates by default
1245 	 * - During streaming, queues via steer() or followUp() based on streamingBehavior option
1246 	 * - Validates model and API key before sending (when not streaming)
1247 	 * @throws Error if streaming and no streamingBehavior specified
1248 	 * @throws Error if no model selected or no API key available (when not streaming)
1249 	 */

1252 		const preflightResult = options?.preflightResult;

1254 
1255 		try {
1256 			// Handle extension commands first (execute immediately, even during streaming)
1257 			// Extension commands manage their own LLM interaction via pi.sendMessage()
1258 			if (expandPromptTemplates && text.startsWith("/")) {
1259 				const handled = await this._tryExecuteExtensionCommand(text);
1260 				if (handled) {
1261 					// Extension command executed, no prompt to send
1262 					preflightResult?.(true);
1263 					return;
1264 				}
1265 			}
1266 
1267 			if (this._compactionAbortController !== undefined) {
1268 				throw new Error(
1269 					"Cannot submit a prompt while compaction is in progress. Wait for compaction to finish and retry.",
1270 				);
1271 			}
1272 
1273 			// Emit input event for extension interception (before skill/template expansion)
1274 			const processedInput = await this._runInputHandlers(
1275 				text,
1276 				options?.images,
1277 				options?.source ?? "interactive",
1278 				this.isStreaming ? options?.streamingBehavior : undefined,
1279 			);
1280 			if (!processedInput) {
1281 				preflightResult?.(true);
1282 				return;
1283 			}

1285

1290 				expandedText = expandPromptTemplate(expandedText, [...this.promptTemplates]);
1291 			}
1292 
1293 			// If streaming, queue via steer() or followUp() based on option
1294 			if (this.isStreaming) {
1295 				if (!options?.streamingBehavior) {
1296 					throw new Error(
1297 						"Agent is already processing. Specify streamingBehavior ('steer' or 'followUp') to queue the message.",
1298 					);
1299 				}
1300 				if (options.streamingBehavior === "followUp") {
1301 					await this._queueFollowUp(expandedText, currentImages);
1302 				} else {
1303 					await this._queueSteer(expandedText, currentImages);
1304 				}
1305 				preflightResult?.(true);
1306 				return;
1307 			}
1308 
1309 			// Flush any pending bash and custom messages before the new prompt
1310 			this._flushPendingBashMessages();
1311 			this._flushPendingCustomMessages();
1312 
1313 			// Validate model
1314 			if (!this.model) {
1315 				throw new Error(formatNoModelSelectedMessage());
1316 			}
1317 
1318 			const hasConfiguredAuth =
1319 				this._modelRuntime.hasConfiguredAuth(this.model.provider) ||
1320 				(await this._modelRuntime.checkAuth(this.model.provider)) !== undefined;
1321 			if (!hasConfiguredAuth) {

1323 				if (isOAuth) {
1324 					throw new Error(

1326 							`Credentials may have expired or network is unavailable. ` +
1327 							`Run '/login ${this.model.provider}' to re-authenticate.`,
1328 					);
1329 				}
1330 				throw new Error(formatNoApiKeyFoundMessage(this.model.provider));
1331 			}
1332 
1333 			// Check if we need to compact before sending (catches aborted responses).
1334 			// The user's new prompt is sent below, so do not call agent.continue() here.
1335 			const lastAssistant = this._findLastAssistantMessage();
1336 			if (lastAssistant) {
1337 				await this._checkCompaction(lastAssistant, false);
1338 			}
1339 
1340 			// Build messages array (custom message if any, then user message)
1341 			messages = [];
1342 
1343 			// Add user message
1344 			const userContent: (TextContent | ImageContent)[] = [{ type: "text", text: expandedText }];
1345 			if (currentImages) {
1346 				userContent.push(...currentImages);
1347 			}
1348 			messages.push({
1349 				role: "user",
1350 				content: userContent,
1351 				timestamp: Date.now(),
1352 			});
1353

1357 			}
1358 			this._pendingNextTurnMessages = [];

1361 			const selectedToolsBefore = this._baseSystemPromptOptions.selectedTools;

1363 				expandedText,
1364 				currentImages,
1365 				this._baseSystemPromptOptions,
1366 			);
1367 			// Handlers may edit event.systemPromptOptions.selectedTools or call setActiveTools(),
1368 			// which updates the live loadout instead. An explicit edit wins; otherwise the live
1369 			// loadout is authoritative, so a setActiveTools() call is not undone here.
1370 			const handlerEditedTools =
1371 				result.systemPromptOptions.selectedTools.length !== selectedToolsBefore.length ||
1372 				result.systemPromptOptions.selectedTools.some((name, index) => name !== selectedToolsBefore[index]);
1373 			if (!handlerEditedTools) result.systemPromptOptions.selectedTools = this.getActiveToolNames();

1375 				messages.push({
1376 					role: "custom",
1377 					customType: msg.customType,
1378 					// Untyped extensions can pass null/missing content; normalize at ingestion.
1379 					content: msg.content ?? [],
1380 					display: msg.display,
1381 					details: msg.details,
1382 					timestamp: Date.now(),
1383 				});
1384 			}
1385 			const updateMessage = this._preparePromptAndToolLoadout(result.systemPromptOptions);
1386 			this._runSystemPromptOptions = result.systemPromptOptions;
1387 			if (updateMessage) messages.unshift(updateMessage);
1388 		} catch (error) {
1389 			preflightResult?.(false);
1390 			throw error;
1391 		}
1392 
1393 		if (!messages) {
1394 			return;
1395 		}
1396 
1397 		preflightResult?.(true);
1398 		await this._runAgentPrompt(messages);
1399 	}

1401 	/**
1402 	 * Try to execute an extension command. Returns true if command was found and executed.
1403 	 */
1404 	private async _tryExecuteExtensionCommand(text: string): Promise<boolean> {
1405 		// Parse command name and args
1406 		const spaceIndex = text.indexOf(" ");
1407 		const commandName = spaceIndex === -1 ? text.slice(1) : text.slice(1, spaceIndex);
1408 		const args = spaceIndex === -1 ? "" : text.slice(spaceIndex + 1);
1409 
1410 		const command = this._extensionRunner.getCommand(commandName);
1411 		if (!command) return false;
1412 
1413 		// Get command context from extension runner (includes session control methods)
1414 		const ctx = this._extensionRunner.createCommandContext();
1415 
1416 		try {
1417 			await command.handler(args, ctx);
1418 			return true;
1419 		} catch (err) {
1420 			// Emit error via extension runner
1421 			this._extensionRunner.emitError({
1422 				extensionPath: `command:${commandName}`,
1423 				event: "command",
1424 				error: err instanceof Error ? err.message : String(err),
1425 			});
1426 			return true;
1427 		}

1430 	/**
1431 	 * Expand skill commands (/skill:name args) to their full content.
1432 	 * Returns the expanded text, or the original text if not a skill command or skill not found.
1433 	 * Emits errors via extension runner if file read fails.
1434 	 */
1435 	private _expandSkillCommand(text: string): string {
1436 		if (!text.startsWith("/skill:")) return text;
1437 
1438 		const spaceIndex = text.indexOf(" ");
1439 		const skillName = spaceIndex === -1 ? text.slice(7) : text.slice(7, spaceIndex);
1440 		const args = spaceIndex === -1 ? "" : text.slice(spaceIndex + 1).trim();
1441 
1442 		const skill = this.resourceLoader.getSkills().skills.find((s) => s.name === skillName);
1443 		if (!skill) return text; // Unknown skill, pass through
1444 
1445 		try {
1446 			const content = readFileSync(skill.filePath, "utf-8");
1447 			const body = stripFrontmatter(content).trim();
1448 			const skillBlock = `<skill name="${skill.name}" location="${skill.filePath}">\nReferences are relative to ${skill.baseDir}.\n\n${body}\n</skill>`;
1449 			return args ? `${skillBlock}\n\n${args}` : skillBlock;
1450 		} catch (err) {
1451 			// Emit error like extension commands do
1452 			this._extensionRunner.emitError({
1453 				extensionPath: skill.filePath,
1454 				event: "skill_expansion",
1455 				error: err instanceof Error ? err.message : String(err),
1456 			});
1457 			return text; // Return original on error
1458 		}
1459 	}

1567 	 * - Streaming: queues message, processed when loop pulls from queue
1568 	 * - Streaming + triggerTurn false: appended to state/session once the current turn ends
1569 	 * - Not streaming + triggerTurn: appends to state/session, starts new turn
1570 	 * - Not streaming + no trigger: appends to state/session, no turn
1571 	 *
1572 	 * @param message Custom message with customType, content, display, details
1573 	 * @param options.triggerTurn If true and not streaming, triggers a new LLM turn
1574 	 * @param options.deliverAs Delivery mode: "steer", "followUp", or "nextTurn"
1575 	 */
1576 	async sendCustomMessage<T = unknown>(
1577 		message: Pick<CustomMessage<T>, "customType" | "content" | "display" | "details">,
1578 		options?: { triggerTurn?: boolean; deliverAs?: "steer" | "followUp" | "nextTurn" },
1579 	): Promise<void> {
1580 		const appMessage = {
1581 			role: "custom" as const,
1582 			customType: message.customType,
1583 			// Untyped extensions can pass null/missing content; normalize at ingestion.
1584 			content: message.content ?? [],
1585 			display: message.display,
1586 			details: message.details,
1587 			timestamp: Date.now(),
1588 		} satisfies CustomMessage<T>;
1589 		if (options?.deliverAs === "nextTurn") {
1590 			this._pendingNextTurnMessages.push(appMessage);
1591 		} else if (this.isStreaming && options?.triggerTurn !== false) {
1592 			if (options?.deliverAs === "followUp") {
1593 				this.agent.followUp(appMessage);
1594 			} else {
1595 				this.agent.steer(appMessage);
1596 			}
1597 		} else if (options?.triggerTurn) {
1598 			await this._runAgentPrompt(appMessage);
1599 		} else if (this.isStreaming) {
1600 			// Appending now would put the message between an assistant tool call and its
1601 			// result, which providers that validate message order reject on replay. Defer
1602 			// to the end of the turn. Nothing is emitted yet: message events must not
1603 			// describe messages the session tree does not contain.
1604 			this._pendingCustomMessages.push(appMessage);
1605 		} else {
1606 			this._appendCustomMessage(appMessage);
1607 		}
1608 	}
