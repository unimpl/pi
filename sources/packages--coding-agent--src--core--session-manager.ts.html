 343 function buildSessionPath(
 344 	entries: SessionEntry[],
 345 	leafId?: string | null,
 346 	byId?: Map<string, SessionEntry>,
 347 ): SessionEntry[] {
 348 	const index = buildEntryIndex(entries, byId);
 349 	let leaf: SessionEntry | undefined;
 350 	if (leafId === null) {
 351 		return [];
 352 	}
 353 	if (leafId) {
 354 		leaf = index.get(leafId);
 355 	}
 356 	leaf ??= entries[entries.length - 1];
 357 	if (!leaf) {
 358 		return [];
 359 	}
 360 
 361 	const path: SessionEntry[] = [];
 362 	let current: SessionEntry | undefined = leaf;
 363 	while (current) {
 364 		path.push(current);
 365 		current = current.parentId ? index.get(current.parentId) : undefined;
 366 	}
 367 	path.reverse();
 368 	return path;

 393 	if (entry.type === "message") {

 395 		// Session files are parsed without validation; old versions, forks, or

 410 	}

 415 		const summary = createCompactionSummaryMessage(entry.summary, entry.tokensBefore, entry.timestamp);
 416 		return entry.systemMessage ? [entry.systemMessage, summary] : [summary];

 421 /**
 422  * Build the active, compaction-aware session entry list.
 423  *
 424  * This follows the current leaf path. If the path contains compaction entries,
 425  * the latest compaction is represented by the compaction entry itself, followed
 426  * by the kept entries starting at firstKeptEntryId and all entries after the
 427  * compaction entry. Older summarized entries are omitted.
 428  */
 429 export function buildContextEntries(
 430 	entries: SessionEntry[],
 431 	leafId?: string | null,
 432 	byId?: Map<string, SessionEntry>,
 433 ): SessionEntry[] {
 434 	const path = buildSessionPath(entries, leafId, byId);
 435 	let compaction: CompactionEntry | null = null;
 436 
 437 	for (const entry of path) {
 438 		if (entry.type === "compaction") {
 439 			compaction = entry;
 440 		}
 441 	}
 442 
 443 	if (!compaction) {
 444 		return path;
 445 	}
 446 
 447 	const compactionIdx = path.findIndex((entry) => entry.id === compaction.id);
 448 	if (compactionIdx < 0) {
 449 		return path;
 450 	}
 451 
 452 	const contextEntries: SessionEntry[] = [compaction];
 453 	let foundFirstKept = false;
 454 	for (let i = 0; i < compactionIdx; i++) {
 455 		const entry = path[i];
 456 		if (entry.id === compaction.firstKeptEntryId) {
 457 			foundFirstKept = true;
 458 		}
 459 		if (foundFirstKept && !(entry.type === "message" && entry.message.role === "system")) {
 460 			contextEntries.push(entry);
 461 		}
 462 	}
 463 	contextEntries.push(...path.slice(compactionIdx + 1));
 464 	return contextEntries;
 465 }

 524 /** Exported for testing */
 525 export function loadEntriesFromFile(filePath: string): FileEntry[] {
 526 	const resolvedFilePath = normalizePath(filePath);
 527 	if (!existsSync(resolvedFilePath)) return [];
 528 
 529 	const entries: FileEntry[] = [];
 530 	let pending = "";
 531 	const fd = openSync(resolvedFilePath, "r");
 532 	try {
 533 		const decoder = new StringDecoder("utf8");
 534 		const buffer = Buffer.allocUnsafe(SESSION_READ_BUFFER_SIZE);
 535 
 536 		while (true) {
 537 			const bytesRead = readSync(fd, buffer, 0, buffer.length, null);
 538 			if (bytesRead === 0) break;
 539 
 540 			pending += decoder.write(buffer.subarray(0, bytesRead));
 541 			let lineStart = 0;
 542 			let newlineIndex = pending.indexOf("\n", lineStart);
 543 			while (newlineIndex !== -1) {
 544 				const entry = parseSessionEntryLine(pending.slice(lineStart, newlineIndex));
 545 				if (entry) entries.push(entry);
 546 				lineStart = newlineIndex + 1;
 547 				newlineIndex = pending.indexOf("\n", lineStart);
 548 			}
 549 			pending = pending.slice(lineStart);
 550 		}
 551 
 552 		pending += decoder.end();
 553 		const finalEntry = parseSessionEntryLine(pending);
 554 		if (finalEntry) entries.push(finalEntry);
 555 	} finally {
 556 		closeSync(fd);
 557 	}
 558 
 559 	// Validate session header before repairing the file.
 560 	if (entries.length === 0) return entries;
 561 	const header = entries[0];
 562 	if (header.type !== "session" || typeof (header as { id?: unknown }).id !== "string") {
 563 		return [];
 564 	}
 565 
 566 	if (pending) appendFileSync(resolvedFilePath, "\n");
 567 	return entries;
 568 }

 583 function readSessionHeader(filePath: string): SessionHeader | null {
 584 	const fd = openSync(filePath, "r");
 585 	try {
 586 		const decoder = new StringDecoder("utf8");
 587 		const buffer = Buffer.allocUnsafe(SESSION_HEADER_READ_BUFFER_SIZE);
 588 		const lineChunks: string[] = [];
 589 		let scannedBytes = 0;
 590 
 591 		while (scannedBytes < MAX_SESSION_HEADER_SCAN_BYTES) {
 592 			const readLength = Math.min(buffer.length, MAX_SESSION_HEADER_SCAN_BYTES - scannedBytes);
 593 			const bytesRead = readSync(fd, buffer, 0, readLength, null);
 594 			if (bytesRead === 0) {
 595 				lineChunks.push(decoder.end());
 596 				return parseSessionHeaderCandidate(lineChunks.join("")) ?? null;
 597 			}
 598 			scannedBytes += bytesRead;
 599 
 600 			const chunk = decoder.write(buffer.subarray(0, bytesRead));
 601 			let lineStart = 0;
 602 			let newlineIndex = chunk.indexOf("\n", lineStart);
 603 			while (newlineIndex !== -1) {
 604 				lineChunks.push(chunk.slice(lineStart, newlineIndex));
 605 				const header = parseSessionHeaderCandidate(lineChunks.join(""));
 606 				if (header !== undefined) return header;
 607 				lineChunks.length = 0;
 608 				lineStart = newlineIndex + 1;
 609 				newlineIndex = chunk.indexOf("\n", lineStart);
 610 			}
 611 			lineChunks.push(chunk.slice(lineStart));
 612 		}
 613 
 614 		// Probe for EOF so a final header without a newline is allowed when it ends
 615 		// exactly at the scan limit. Any additional byte exceeds the bounded scan.
 616 		const probe = Buffer.allocUnsafe(1);
 617 		if (readSync(fd, probe, 0, probe.length, null) === 0) {
 618 			lineChunks.push(decoder.end());
 619 			return parseSessionHeaderCandidate(lineChunks.join("")) ?? null;
 620 		}
 621 		throw new SessionHeaderScanLimitError(filePath);
 622 	} finally {
 623 		closeSync(fd);
 624 	}
 625 }

 699 async function buildSessionInfo(filePath: string): Promise<SessionInfo | null> {
 700 	try {
 701 		const stats = await stat(filePath);
 702 		let header: SessionHeader | null = null;
 703 		let messageCount = 0;
 704 		let firstMessage = "";
 705 		const allMessages: string[] = [];
 706 		let name: string | undefined;
 707 		let lastActivityTime: number | undefined;
 708 
 709 		const rl = createInterface({
 710 			input: createReadStream(filePath, { encoding: "utf8" }),
 711 			crlfDelay: Infinity,
 712 		});
 713 
 714 		for await (const line of rl) {
 715 			const entry = parseSessionEntryLine(line);
 716 			if (!entry) continue;
 717 
 718 			if (!header) {
 719 				if (entry.type !== "session") return null;
 720 				header = entry;
 721 				continue;
 722 			}
 723 
 724 			// Extract session name (use latest, including explicit clears)
 725 			if (entry.type === "session_info") {
 726 				name = entry.name?.trim() || undefined;
 727 			}
 728 
 729 			if (entry.type !== "message") continue;
 730 			messageCount++;
 731 
 732 			const activityTime = getMessageActivityTime(entry);
 733 			if (typeof activityTime === "number") {
 734 				lastActivityTime = Math.max(lastActivityTime ?? 0, activityTime);
 735 			}
 736 
 737 			const message = entry.message;
 738 			if (!isMessageWithContent(message)) continue;
 739 			if (message.role !== "user" && message.role !== "assistant") continue;
 740 
 741 			const textContent = extractTextContent(message);
 742 			if (!textContent) continue;
 743 
 744 			allMessages.push(textContent);
 745 			if (!firstMessage && message.role === "user") {
 746 				firstMessage = textContent;
 747 			}
 748 		}
 749 
 750 		if (!header) return null;
 751 
 752 		const cwd = typeof header.cwd === "string" ? header.cwd : "";
 753 		const parentSessionPath = header.parentSession;
 754 		const headerTime = typeof header.timestamp === "string" ? new Date(header.timestamp).getTime() : NaN;
 755 		const modified =
 756 			typeof lastActivityTime === "number" && lastActivityTime > 0
 757 				? new Date(lastActivityTime)
 758 				: !Number.isNaN(headerTime)
 759 					? new Date(headerTime)
 760 					: stats.mtime;
 761 
 762 		return {
 763 			path: filePath,
 764 			id: header.id,
 765 			cwd,
 766 			name,
 767 			parentSessionPath,
 768 			created: new Date(header.timestamp),
 769 			modified,
 770 			messageCount,
 771 			firstMessage: firstMessage || "(no messages)",
 772 			allMessagesText: allMessages.join(" "),
 773 		};
 774 	} catch {
 775 		return null;
 776 	}
 777 }

 782 
 783 async function buildSessionInfosWithConcurrency(
 784 	files: string[],
 785 	onLoaded: () => void,
 786 ): Promise<(SessionInfo | null)[]> {
 787 	const results: (SessionInfo | null)[] = new Array(files.length).fill(null);
 788 	const inFlight = new Set<Promise<void>>();
 789 	let nextIndex = 0;
 790 
 791 	const startNext = (): void => {
 792 		const index = nextIndex++;
 793 		const file = files[index];
 794 		if (!file) return;
 795 
 796 		let task: Promise<void>;
 797 		task = buildSessionInfo(file)
 798 			.then((info) => {
 799 				results[index] = info;
 800 			})
 801 			.catch(() => {
 802 				results[index] = null;
 803 			})
 804 			.finally(() => {
 805 				inFlight.delete(task);
 806 				onLoaded();
 807 			});
 808 		inFlight.add(task);
 809 	};
 810 
 811 	while (nextIndex < files.length || inFlight.size > 0) {
 812 		while (nextIndex < files.length && inFlight.size < MAX_CONCURRENT_SESSION_INFO_LOADS) {
 813 			startNext();
 814 		}
 815 		if (inFlight.size > 0) {
 816 			await Promise.race(inFlight);
 817 		}
 818 	}
 819 
 820 	return results;
 821 }

 823 async function listSessionsFromDir(
 824 	dir: string,
 825 	onProgress?: SessionListProgress,
 826 	progressOffset = 0,
 827 	progressTotal?: number,

 831 		return sessions;
 832 	}

 856 /**
 857  * Manages conversation sessions as append-only trees stored in JSONL files.
 858  *
 859  * Each session entry has an id and parentId forming a tree structure. The "leaf"
 860  * pointer tracks the current position. Appending creates a child of the current leaf.
 861  * Branching moves the leaf to an earlier entry, allowing new branches without
 862  * modifying history.
 863  *
 864  * Use buildSessionContext() to get the resolved message list for the LLM, which
 865  * handles compaction summaries and follows the path from root to current leaf.
 866  */
 867 export class SessionManager {
 868 	private sessionId: string = "";
 869 	private sessionFile: string | undefined;
 870 	private sessionDir: string;
 871 	private cwd: string;
 872 	private persist: boolean;
 873 	private flushed: boolean = false;
 874 	private fileEntries: FileEntry[] = [];
 875 	private byId: Map<string, SessionEntry> = new Map();
 876 	private labelsById: Map<string, string> = new Map();
 877 	private labelTimestampsById: Map<string, string> = new Map();
 878 	private leafId: string | null = null;
 879 
 880 	private constructor(
 881 		cwd: string,
 882 		sessionDir: string,
 883 		sessionFile: string | undefined,
 884 		persist: boolean,
 885 		newSessionOptions?: NewSessionOptions,
 886 		preloadedFileEntries?: FileEntry[],
 887 	) {
 888 		this.cwd = resolvePath(cwd);
 889 		this.sessionDir = normalizePath(sessionDir);
 890 		this.persist = persist;
 891 		if (persist && this.sessionDir && !existsSync(this.sessionDir)) {
 892 			mkdirSync(this.sessionDir, { recursive: true });
 893 		}
 894 
 895 		if (sessionFile) {
 896 			this._setSessionFile(sessionFile, preloadedFileEntries);
 897 		} else if (preloadedFileEntries?.length) {
 898 			this._loadEntries(preloadedFileEntries, newSessionOptions);
 899 		} else {
 900 			this.newSession(newSessionOptions);
 901 		}
 902 	}

 909 	private _setSessionFile(sessionFile: string, preloadedFileEntries?: FileEntry[]): void {
 910 		this.sessionFile = resolvePath(sessionFile);
 911 		if (existsSync(this.sessionFile)) {
 912 			const entries = preloadedFileEntries ?? loadEntriesFromFile(this.sessionFile);
 913 
 914 			// If file was empty, initialize it with a valid session header. If it was
 915 			// non-empty but did not parse as a pi session, fail without modifying it.
 916 			if (entries.length === 0) {
 917 				const explicitPath = this.sessionFile;
 918 				if (statSync(explicitPath).size > 0) {
 919 					throw new Error(`Session file is not a valid ${APP_NAME} session: ${explicitPath}`);
 920 				}
 921 				this.newSession();
 922 				this.sessionFile = explicitPath;
 923 				this._rewriteFile();
 924 				this.flushed = true;
 925 				return;
 926 			}
 927 
 928 			this._loadEntries(entries);
 929 			this.flushed = true;
 930 		} else {
 931 			const explicitPath = this.sessionFile;
 932 			this.newSession();
 933 			this.sessionFile = explicitPath; // preserve explicit path from --session flag
 934 		}
 935 	}

 944 			type: "session",
 945 			version: CURRENT_SESSION_VERSION,
 946 			id: this.sessionId,
 947 			timestamp,
 948 			cwd: this.cwd,
 949 			parentSession: options?.parentSession,
 950 		};

 952 		this.byId.clear();
 953 		this.labelsById.clear();
 954 		this.labelTimestampsById.clear();
 955 		this.leafId = null;
 956 		this.flushed = false;
 957

 959 			const fileTimestamp = timestamp.replace(/[:.]/g, "-");
 960 			this.sessionFile = join(this.getSessionDir(), `${fileTimestamp}_${this.sessionId}.jsonl`);
 961 		}
 962 		return this.sessionFile;
 963 	}

 965 	private _loadEntries(entries: FileEntry[], options?: NewSessionOptions): void {
 966 		const header = entries.find((e) => e.type === "session") as SessionHeader | undefined;
 967 
 968 		if (header) {
 969 			this.fileEntries = entries;
 970 			this.sessionId = header.id;
 971 
 972 			if (migrateToCurrentVersion(this.fileEntries)) {
 973 				this._rewriteFile();
 974 			}
 975 		} else {
 976 			this.newSession(options);
 977 			this.fileEntries = this.fileEntries.concat(entries);
 978 		}
 979 
 980 		this._buildIndex();
 981 	}

 983 	private _buildIndex(): void {
 984 		this.byId.clear();
 985 		this.labelsById.clear();
 986 		this.labelTimestampsById.clear();
 987 		this.leafId = null;
 988 		for (const entry of this.fileEntries) {
 989 			if (entry.type === "session") continue;
 990 			this.byId.set(entry.id, entry);
 991 			this.leafId = entry.id;
 992 			if (entry.type === "label") {
 993 				if (entry.label) {
 994 					this.labelsById.set(entry.targetId, entry.label);
 995 					this.labelTimestampsById.set(entry.targetId, entry.timestamp);
 996 				} else {
 997 					this.labelsById.delete(entry.targetId);
 998 					this.labelTimestampsById.delete(entry.targetId);
 999 				}
1000 			}
1001 		}
1002 	}

1062 			}

1333 	/**
1334 	 * Get the session as a tree structure. Returns a shallow defensive copy of all entries.
1335 	 * A well-formed session has exactly one root (first entry with parentId === null).
1336 	 * Orphaned entries (broken parent chain) are also returned as roots.
1337 	 */
1338 	getTree(): SessionTreeNode[] {
1339 		const entries = this.getEntries();
1340 		const nodeMap = new Map<string, SessionTreeNode>();
1341 		const roots: SessionTreeNode[] = [];
1342 
1343 		// Create nodes with resolved labels
1344 		for (const entry of entries) {
1345 			const label = this.labelsById.get(entry.id);
1346 			const labelTimestamp = this.labelTimestampsById.get(entry.id);
1347 			nodeMap.set(entry.id, { entry, children: [], label, labelTimestamp });
1348 		}
1349 
1350 		// Build tree
1351 		for (const entry of entries) {
1352 			const node = nodeMap.get(entry.id)!;
1353 			if (entry.parentId === null || entry.parentId === entry.id) {
1354 				roots.push(node);
1355 			} else {
1356 				const parent = nodeMap.get(entry.parentId);
1357 				if (parent) {
1358 					parent.children.push(node);
1359 				} else {
1360 					// Orphan - treat as root
1361 					roots.push(node);
1362 				}
1363 			}
1364 		}
1365 
1366 		// Sort children by timestamp (oldest first, newest at bottom)
1367 		// Use iterative approach to avoid stack overflow on deep trees
1368 		const stack: SessionTreeNode[] = [...roots];
1369 		while (stack.length > 0) {
1370 			const node = stack.pop()!;
1371 			node.children.sort((a, b) => new Date(a.entry.timestamp).getTime() - new Date(b.entry.timestamp).getTime());
1372 			stack.push(...node.children);
1373 		}
1374 
1375 		return roots;
1376 	}

1450 		// removing labels requires re-chaining the retained path to avoid orphaned subtrees.
1451 		const pathWithoutLabels: SessionEntry[] = [];
1452 		const replacementByLabelId = new Map<string, string>();
1453 		const pendingLabelIds: string[] = [];
1454 		let pathParentId: string | null = null;
1455 		for (const entry of path) {
1456 			if (entry.type === "label") {
1457 				pendingLabelIds.push(entry.id);
1458 				continue;
1459 			}
1460 			for (const labelId of pendingLabelIds) {
1461 				replacementByLabelId.set(labelId, entry.id);
1462 			}
1463 			pendingLabelIds.length = 0;
1464 			pathWithoutLabels.push(
1465 				entry.type === "compaction"
1466 					? {
1467 							...entry,
1468 							parentId: pathParentId,
1469 							firstKeptEntryId: replacementByLabelId.get(entry.firstKeptEntryId) ?? entry.firstKeptEntryId,
1470 						}
1471 					: { ...entry, parentId: pathParentId },
1472 			);
1473 			pathParentId = entry.id;
1474 		}
1475 
1476 		const newSessionId = createSessionId();
1477 		const timestamp = new Date().toISOString();
1478 		const fileTimestamp = timestamp.replace(/[:.]/g, "-");

1482 			type: "session",
1483 			version: CURRENT_SESSION_VERSION,
1484 			id: newSessionId,
1485 			timestamp,
1486 			cwd: this.cwd,
1487 			parentSession: this.persist ? previousSessionFile : undefined,
1488 		};
1489 
1490 		// Collect labels for entries in the path
1491 		const pathEntryIds = new Set(pathWithoutLabels.map((e) => e.id));
1492 		const labelsToWrite: Array<{ targetId: string; label: string; timestamp: string }> = [];
1493 		for (const [targetId, label] of this.labelsById) {
1494 			if (pathEntryIds.has(targetId)) {
1495 				labelsToWrite.push({ targetId, label, timestamp: this.labelTimestampsById.get(targetId)! });
1496 			}
1497 		}
1498 
1499 		if (this.persist) {
1500 			// Build label entries
1501 			const lastEntryId = pathWithoutLabels[pathWithoutLabels.length - 1]?.id || null;
1502 			let parentId = lastEntryId;
1503 			const labelEntries: LabelEntry[] = [];
1504 			for (const { targetId, label, timestamp: labelTimestamp } of labelsToWrite) {
1505 				const labelEntry: LabelEntry = {
1506 					type: "label",
1507 					id: generateId(new Set(pathEntryIds)),
1508 					parentId,
1509 					timestamp: labelTimestamp,
1510 					targetId,
1511 					label,
1512 				};
1513 				pathEntryIds.add(labelEntry.id);
1514 				labelEntries.push(labelEntry);
1515 				parentId = labelEntry.id;
1516 			}
1517 
1518 			this.fileEntries = [header, ...pathWithoutLabels, ...labelEntries];
1519 			this.sessionId = newSessionId;
1520 			this.sessionFile = newSessionFile;
1521 			this._buildIndex();
1522 
1523 			// Only write the file now if it contains an assistant message.
1524 			// Otherwise defer to _persist(), which creates the file on the
1525 			// first assistant response, matching the newSession() contract
1526 			// and avoiding the duplicate-header bug when _persist()'s
1527 			// no-assistant guard later resets flushed to false.
1528 			const hasAssistant = this.fileEntries.some((e) => e.type === "message" && e.message.role === "assistant");
1529 			if (hasAssistant) {
1530 				this._rewriteFile();
1531 				this.flushed = true;
1532 			} else {
1533 				this.flushed = false;
1534 			}
1535 
1536 			return newSessionFile;
1537 		}
1538 
1539 		// In-memory mode: replace current session with the path + labels
1540 		const labelEntries: LabelEntry[] = [];
1541 		let parentId = pathWithoutLabels[pathWithoutLabels.length - 1]?.id || null;
1542 		for (const { targetId, label, timestamp: labelTimestamp } of labelsToWrite) {
1543 			const labelEntry: LabelEntry = {
1544 				type: "label",
1545 				id: generateId(new Set([...pathEntryIds, ...labelEntries.map((e) => e.id)])),
1546 				parentId,
1547 				timestamp: labelTimestamp,
1548 				targetId,
1549 				label,
1550 			};
1551 			labelEntries.push(labelEntry);
1552 			parentId = labelEntry.id;
1553 		}
1554 		this.fileEntries = [header, ...pathWithoutLabels, ...labelEntries];
1555 		this.sessionId = newSessionId;
1556 		this._buildIndex();
1557 		return undefined;
1558 	}
