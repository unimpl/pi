 149 		this.messages.push(message);

 153 		return this.messages.length > 0;

 156 	drain(): AgentMessage[] {
 157 		if (this.mode === "all") {
 158 			const drained = this.messages.slice();

 161 		}

 167 		this.messages = this.messages.slice(1);

 169 	}

 376 	/** Continue from the current transcript. The last message must be a user or tool-result message. */
 377 	async continue(): Promise<void> {
 378 		if (this.activeRun) {
 379 			throw new Error("Agent is already processing. Wait for completion before continuing.");
 380 		}
 381 
 382 		const lastMessage = this._state.messages[this._state.messages.length - 1];
 383 		if (!lastMessage || this._state.messages.every((message) => message.role === "system")) {
 384 			throw new Error("No messages to continue from");
 385 		}
 386 
 387 		if (lastMessage.role === "assistant") {
 388 			const queuedSteering = this.steeringQueue.drain();
 389 			if (queuedSteering.length > 0) {
 390 				await this.runPromptMessages(queuedSteering, { skipInitialSteeringPoll: true });
 391 				return;
 392 			}
 393 
 394 			const queuedFollowUps = this.followUpQueue.drain();
 395 			if (queuedFollowUps.length > 0) {
 396 				await this.runPromptMessages(queuedFollowUps);
 397 				return;
 398 			}
 399 
 400 			throw new Error("Cannot continue from message role: assistant");
 401 		}
 402 
 403 		await this.runContinuation();
 404 	}

 501 	private async runWithLifecycle(executor: (signal: AbortSignal) => Promise<void>): Promise<void> {
 502 		if (this.activeRun) {
 503 			throw new Error("Agent is already processing.");
 504 		}
 505 
 506 		const abortController = new AbortController();
 507 		let resolvePromise = () => {};
 508 		const promise = new Promise<void>((resolve) => {
 509 			resolvePromise = resolve;
 510 		});
 511 		this.activeRun = { promise, resolve: resolvePromise, abortController };
 512 
 513 		this._state.isStreaming = true;
 514 		this._state.streamingMessage = undefined;
 515 		this._state.errorMessage = undefined;
 516 
 517 		try {
 518 			await executor(abortController.signal);
 519 		} catch (error) {
 520 			await this.handleRunFailure(error, abortController.signal.aborted);
 521 		} finally {
 522 			this.finishRun();
 523 		}
 524 	}

 526 	private async handleRunFailure(error: unknown, aborted: boolean): Promise<void> {
 527 		const failureMessage = {
 528 			role: "assistant",
 529 			content: [{ type: "text", text: "" }],
 530 			api: this._state.model.api,
 531 			provider: this._state.model.provider,
 532 			model: this._state.model.id,
 533 			usage: EMPTY_USAGE,
 534 			stopReason: aborted ? "aborted" : "error",
 535 			errorMessage: error instanceof Error ? error.message : String(error),
 536 			timestamp: Date.now(),
 537 		} satisfies AgentMessage;
 538 		await this.processEvents({ type: "message_start", message: failureMessage });
 539 		await this.processEvents({ type: "message_end", message: failureMessage });
 540 		await this.processEvents({ type: "turn_end", message: failureMessage, toolResults: [] });
 541 		await this.processEvents({ type: "agent_end", messages: [failureMessage] });
 542 	}

 545 		this._state.isStreaming = false;
 546 		this._state.streamingMessage = undefined;
 547 		this._state.pendingToolCalls = new Set<string>();
 548 		this.activeRun?.resolve();
 549 		this.activeRun = undefined;

 552 	/**
 553 	 * Reduce internal state for a loop event, then await listeners.
 554 	 *
 555 	 * `agent_end` only means no further loop events will be emitted. The run is
 556 	 * considered idle later, after all awaited listeners for `agent_end` finish
 557 	 * and `finishRun()` clears runtime-owned state.
 558 	 */
 559 	private async processEvents(event: AgentEvent): Promise<void> {
 560 		switch (event.type) {
 561 			case "message_start":
 562 				this._state.streamingMessage = event.message;
 563 				break;
 564 
 565 			case "message_update":
 566 				this._state.streamingMessage = event.message;
 567 				break;
 568 
 569 			case "message_end":
 570 				this._state.streamingMessage = undefined;
 571 				this._state.messages.push(event.message);
 572 				break;
 573 
 574 			case "tool_execution_start": {
 575 				const pendingToolCalls = new Set(this._state.pendingToolCalls);
 576 				pendingToolCalls.add(event.toolCallId);
 577 				this._state.pendingToolCalls = pendingToolCalls;
 578 				break;
 579 			}
 580 
 581 			case "tool_execution_end": {
 582 				const pendingToolCalls = new Set(this._state.pendingToolCalls);
 583 				pendingToolCalls.delete(event.toolCallId);
 584 				this._state.pendingToolCalls = pendingToolCalls;
 585 				break;
 586 			}
 587 
 588 			case "turn_end":
 589 				if (event.message.role === "assistant" && event.message.errorMessage) {
 590 					this._state.errorMessage = event.message.errorMessage;
 591 				}
 592 				break;
 593 
 594 			case "agent_end":
 595 				this._state.streamingMessage = undefined;
 596 				break;
 597 		}
 598 
 599 		const signal = this.activeRun?.abortController.signal;
 600 		if (!signal) {
 601 			throw new Error("Agent listener invoked outside active run");
 602 		}
 603 		for (const listener of this.listeners) {
 604 			await listener(event, signal);
 605 		}
 606 	}
