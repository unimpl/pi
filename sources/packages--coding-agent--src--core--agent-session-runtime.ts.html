  56 function extractUserMessageText(content: string | Array<{ type: string; text?: string }>): string {
  57 	if (typeof content === "string") {
  58 		return content;
  59 	}
  60 
  61 	return content
  62 		.filter((part): part is { type: "text"; text: string } => part.type === "text" && typeof part.text === "string")
  63 		.map((part) => part.text)
  64 		.join("");
  65 }

 196 	async switchSession(
 197 		sessionPath: string,
 198 		options?: {
 199 			cwdOverride?: string;
 200 			withSession?: (ctx: ReplacedSessionContext) => Promise<void>;
 201 			projectTrustContextFactory?: (cwd: string) => ProjectTrustContext;
 202 		},
 203 	): Promise<{ cancelled: boolean }> {
 204 		const beforeResult = await this.emitBeforeSwitch("resume", sessionPath);
 205 		if (beforeResult.cancelled) {
 206 			return beforeResult;
 207 		}
 208 
 209 		const previousSessionFile = this.session.sessionFile;
 210 		const sessionManager = SessionManager.open(sessionPath, undefined, options?.cwdOverride);
 211 		assertSessionCwdExists(sessionManager, this.cwd);
 212 		await this.teardownCurrent("resume", sessionManager.getSessionFile());
 213 		this.apply(
 214 			await this.createRuntime({

 219 				projectTrustContext: options?.projectTrustContextFactory?.(sessionManager.getCwd()),

 223 		return { cancelled: false };

 262 	async fork(
 263 		entryId: string,
 264 		options?: { position?: "before" | "at"; withSession?: (ctx: ReplacedSessionContext) => Promise<void> },
 265 	): Promise<{ cancelled: boolean; selectedText?: string }> {
 266 		const position = options?.position ?? "before";
 267 		const beforeResult = await this.emitBeforeFork(entryId, { position });
 268 		if (beforeResult.cancelled) {
 269 			return { cancelled: true };
 270 		}
 271 		let targetLeafId: string | null;
 272 		let selectedText: string | undefined;
 273 
 274 		const selectedEntry = this.session.sessionManager.getEntry(entryId);
 275 		if (!selectedEntry) {
 276 			throw new Error("Invalid entry ID for forking");
 277 		}
 278 
 279 		if (position === "at") {
 280 			targetLeafId = selectedEntry.id;
 281 		} else {
 282 			if (selectedEntry.type !== "message" || selectedEntry.message.role !== "user") {
 283 				throw new Error("Invalid entry ID for forking");
 284 			}
 285 			targetLeafId = selectedEntry.parentId;
 286 			selectedText = extractUserMessageText(selectedEntry.message.content);
 287 		}
 288 
 289 		const previousSessionFile = this.session.sessionFile;
 290 		if (this.session.sessionManager.isPersisted()) {
 291 			const currentSessionFile = this.session.sessionFile;
 292 			if (!currentSessionFile) {
 293 				throw new Error("Persisted session is missing a session file");
 294 			}
 295 			const sessionDir = this.session.sessionManager.getSessionDir();
 296 			if (!targetLeafId) {
 297 				const sessionManager = SessionManager.create(this.cwd, sessionDir);
 298 				sessionManager.newSession({ parentSession: currentSessionFile });
 299 				await this.teardownCurrent("fork", sessionManager.getSessionFile());
 300 				this.apply(
 301 					await this.createRuntime({
 302 						cwd: this.cwd,
 303 						agentDir: this.services.agentDir,
 304 						sessionManager,
 305 						sessionStartEvent: { type: "session_start", reason: "fork", previousSessionFile },
 306 					}),
 307 				);
 308 				await this.finishSessionReplacement(options?.withSession);
 309 				return { cancelled: false, selectedText };
 310 			}
 311 
 312 			if (!existsSync(currentSessionFile)) {
 313 				throw new Error(
 314 					"This session has not been saved yet. Wait for the first assistant response before cloning or forking it.",
 315 				);
 316 			}
 317 			const sessionManager = SessionManager.open(currentSessionFile, sessionDir);
 318 			const forkedSessionPath = sessionManager.createBranchedSession(targetLeafId);
 319 			if (!forkedSessionPath) {
 320 				throw new Error("Failed to create forked session");
 321 			}
 322 			await this.teardownCurrent("fork", sessionManager.getSessionFile());
 323 			this.apply(
 324 				await this.createRuntime({
 325 					cwd: sessionManager.getCwd(),
 326 					agentDir: this.services.agentDir,
 327 					sessionManager,
 328 					sessionStartEvent: { type: "session_start", reason: "fork", previousSessionFile },
 329 				}),
 330 			);
 331 			await this.finishSessionReplacement(options?.withSession);
 332 			return { cancelled: false, selectedText };
 333 		}
 334 
 335 		const sessionManager = this.session.sessionManager;
 336 		await this.teardownCurrent("fork", sessionManager.getSessionFile());
 337 		if (!targetLeafId) {
 338 			sessionManager.newSession({ parentSession: previousSessionFile });
 339 		} else {
 340 			sessionManager.createBranchedSession(targetLeafId);
 341 		}
 342 		this.apply(
 343 			await this.createRuntime({
 344 				cwd: this.cwd,
 345 				agentDir: this.services.agentDir,
 346 				sessionManager,
 347 				sessionStartEvent: { type: "session_start", reason: "fork", previousSessionFile },
 348 			}),
 349 		);
 350 		await this.finishSessionReplacement(options?.withSession);
 351 		return { cancelled: false, selectedText };
 352 	}

 354 	/**
 355 	 * Import a session JSONL file and switch runtime state to the imported session.
 356 	 *
 357 	 * @returns `{ cancelled: true }` when cancelled by `session_before_switch`, otherwise `{ cancelled: false }`.
 358 	 * @throws {SessionImportFileNotFoundError} When the input path does not exist.
 359 	 * @throws {MissingSessionCwdError} When the imported session cwd cannot be resolved and no override is provided.
 360 	 */
 361 	async importFromJsonl(inputPath: string, cwdOverride?: string): Promise<{ cancelled: boolean }> {
 362 		const resolvedPath = resolvePath(inputPath);
 363 		if (!existsSync(resolvedPath)) {
 364 			throw new SessionImportFileNotFoundError(resolvedPath);
 365 		}
 366 
 367 		const sessionDir = this.session.sessionManager.getSessionDir();
 368 		if (!existsSync(sessionDir)) {
 369 			mkdirSync(sessionDir, { recursive: true });
 370 		}
 371 
 372 		let destinationPath = join(sessionDir, basename(resolvedPath));
 373 		const sourceAlreadyStored = resolve(destinationPath) === resolvedPath;
 374 		if (!sourceAlreadyStored) {
 375 			const { name, ext } = parse(destinationPath);
 376 			let suffix = 1;
 377 			while (existsSync(destinationPath)) {
 378 				destinationPath = join(sessionDir, `${name}-${suffix++}${ext}`);
 379 			}
 380 		}
 381 		const beforeResult = await this.emitBeforeSwitch("resume", destinationPath);
 382 		if (beforeResult.cancelled) {
 383 			return beforeResult;
 384 		}
 385 
 386 		const previousSessionFile = this.session.sessionFile;
 387 		if (!sourceAlreadyStored) {
 388 			copyFileSync(resolvedPath, destinationPath, constants.COPYFILE_EXCL);
 389 		}
 390 
 391 		const sessionManager = SessionManager.open(destinationPath, sessionDir, cwdOverride);
 392 		assertSessionCwdExists(sessionManager, this.cwd);
 393 		await this.teardownCurrent("resume", sessionManager.getSessionFile());
 394 		this.apply(
 395 			await this.createRuntime({
 396 				cwd: sessionManager.getCwd(),
 397 				agentDir: this.services.agentDir,
 398 				sessionManager,
 399 				sessionStartEvent: { type: "session_start", reason: "resume", previousSessionFile },
 400 			}),
 401 		);
 402 		await this.finishSessionReplacement();
 403 		return { cancelled: false };
 404 	}

 416 /**
 417  * Create the initial runtime from a runtime factory and initial session target.
 418  *
 419  * The same factory is stored on the returned AgentSessionRuntime and reused for
 420  * later /new, /resume, /fork, and import flows.
 421  */
 422 export async function createAgentSessionRuntime(
 423 	createRuntime: CreateAgentSessionRuntimeFactory,
 424 	options: {
 425 		cwd: string;
 426 		agentDir: string;
 427 		sessionManager: SessionManager;
 428 		sessionStartEvent?: SessionStartEvent;
 429 	},
 430 ): Promise<AgentSessionRuntime> {
 431 	assertSessionCwdExists(options.sessionManager, options.cwd);
 432 	const result = await createRuntime(options);
 433 	return new AgentSessionRuntime(
 434 		result.session,
 435 		result.services,
 436 		createRuntime,
 437 		result.diagnostics,
 438 		result.modelFallbackMessage,
 439 	);
