 340 function flushSseEvent(state: SseDecoderState): ServerSentEvent | null {
 341 	if (!state.event && state.data.length === 0) {
 342 		return null;
 343 	}
 344 
 345 	const event: ServerSentEvent = {
 346 		event: state.event,
 347 		data: state.data.join("\n"),
 348 		raw: [...state.raw],
 349 	};
 350 	state.event = null;
 351 	state.data = [];
 352 	state.raw = [];
 353 	return event;

 356 function decodeSseLine(line: string, state: SseDecoderState): ServerSentEvent | null {
 357 	if (line === "") {
 358 		return flushSseEvent(state);
 359 	}
 360 
 361 	state.raw.push(line);
 362 	if (line.startsWith(":")) {
 363 		return null;
 364 	}
 365 
 366 	const delimiterIndex = line.indexOf(":");
 367 	const fieldName = delimiterIndex === -1 ? line : line.slice(0, delimiterIndex);
 368 	let value = delimiterIndex === -1 ? "" : line.slice(delimiterIndex + 1);
 369 	if (value.startsWith(" ")) {
 370 		value = value.slice(1);
 371 	}
 372 
 373 	if (fieldName === "event") {
 374 		state.event = value;
 375 	} else if (fieldName === "data") {
 376 		state.data.push(value);
 377 	}
 378 
 379 	return null;
 380 }

 382 function nextLineBreakIndex(text: string): number {
 383 	const carriageReturnIndex = text.indexOf("\r");
 384 	const newlineIndex = text.indexOf("\n");
 385 	if (carriageReturnIndex === -1) {
 386 		return newlineIndex;
 387 	}
 388 	if (newlineIndex === -1) {
 389 		return carriageReturnIndex;
 390 	}
 391 	return Math.min(carriageReturnIndex, newlineIndex);
 392 }

 394 function consumeLine(text: string): { line: string; rest: string } | null {
 395 	const lineBreakIndex = nextLineBreakIndex(text);
 396 	if (lineBreakIndex === -1) {
 397 		return null;
 398 	}
 399 
 400 	let nextIndex = lineBreakIndex + 1;
 401 	if (text[lineBreakIndex] === "\r" && text[nextIndex] === "\n") {
 402 		nextIndex += 1;
 403 	}
 404 
 405 	return {
 406 		line: text.slice(0, lineBreakIndex),
 407 		rest: text.slice(nextIndex),
 408 	};

 411 async function* iterateSseMessages(
 412 	body: ReadableStream<Uint8Array>,
 413 	signal?: AbortSignal,
 414 ): AsyncGenerator<ServerSentEvent> {
 415 	const reader = body.getReader();
 416 	const decoder = new TextDecoder();
 417 	const state: SseDecoderState = { event: null, data: [], raw: [] };
 418 	let buffer = "";
 419 
 420 	try {
 421 		while (true) {
 422 			if (signal?.aborted) {
 423 				throw new Error("Request was aborted");
 424 			}
 425 
 426 			const { value, done } = await reader.read();
 427 			if (done) {
 428 				break;
 429 			}
 430 
 431 			buffer += decoder.decode(value, { stream: true });
 432 			let consumed = consumeLine(buffer);
 433 			while (consumed) {
 434 				buffer = consumed.rest;
 435 				const event = decodeSseLine(consumed.line, state);
 436 				if (event) {
 437 					yield event;
 438 				}
 439 				consumed = consumeLine(buffer);
 440 			}
 441 		}
 442 
 443 		buffer += decoder.decode();
 444 		let consumed = consumeLine(buffer);
 445 		while (consumed) {
 446 			buffer = consumed.rest;
 447 			const event = decodeSseLine(consumed.line, state);
 448 			if (event) {
 449 				yield event;
 450 			}
 451 			consumed = consumeLine(buffer);
 452 		}
 453 
 454 		if (buffer.length > 0) {
 455 			const event = decodeSseLine(buffer, state);
 456 			if (event) {
 457 				yield event;
 458 			}
 459 		}
 460 
 461 		const trailingEvent = flushSseEvent(state);
 462 		if (trailingEvent) {
 463 			yield trailingEvent;
 464 		}
 465 	} finally {
 466 		reader.releaseLock();
 467 	}
 468 }

 470 async function* iterateAnthropicEvents(
 471 	response: Response,
 472 	signal?: AbortSignal,
 473 ): AsyncGenerator<RawMessageStreamEvent> {
 474 	if (!response.body) {
 475 		throw new Error("Attempted to iterate over an Anthropic response with no body");
 476 	}
 477 
 478 	let sawMessageStart = false;
 479 	let sawMessageEnd = false;
 480 
 481 	for await (const sse of iterateSseMessages(response.body, signal)) {
 482 		if (sse.event === "error") {
 483 			throw new Error(sse.data);
 484 		}
 485 
 486 		if (!ANTHROPIC_MESSAGE_EVENTS.has(sse.event ?? "")) {
 487 			continue;
 488 		}
 489 
 490 		try {
 491 			const event = parseJsonWithRepair<RawMessageStreamEvent>(sse.data);
 492 			if (event.type === "message_start") {
 493 				sawMessageStart = true;
 494 			} else if (event.type === "message_stop") {
 495 				sawMessageEnd = true;
 496 			}
 497 			yield event;
 498 		} catch (error) {
 499 			const message = error instanceof Error ? error.message : String(error);
 500 			throw new Error(
 501 				`Could not parse Anthropic SSE event ${sse.event}: ${message}; data=${sse.data}; raw=${sse.raw.join("\\n")}`,
 502 			);
 503 		}
 504 	}
 505 
 506 	if (sawMessageStart && !sawMessageEnd) {
 507 		throw new Error("Anthropic stream ended before message_stop");
 508 	}
 509 }

 619 					output.usage.cacheRead = event.message.usage.cache_read_input_tokens || 0;
 620 					output.usage.cacheWrite = event.message.usage.cache_creation_input_tokens || 0;
 621 					output.usage.cacheWrite1h = event.message.usage.cache_creation?.ephemeral_1h_input_tokens || 0;
 622 					// Anthropic doesn't provide total_tokens, compute from components
 623 					output.usage.totalTokens =
 624 						output.usage.input + output.usage.output + output.usage.cacheRead + output.usage.cacheWrite;
 625 					calculateCost(usageModel, output.usage);
 626 				} else if (event.type === "content_block_start") {
 627 					if (event.content_block.type === "fallback") {
 628 						if (output.content.length > 0) {
 629 							throw new Error("Anthropic performed an unsupported mid-output model fallback");
 630 						}
 631 						continue;
 632 					}
 633 					if (event.content_block.type === "text") {
 634 						const block: Block = {
 635 							type: "text",
 636 							text: event.content_block.text ?? "",
 637 							index: event.index,
 638 						};
 639 						output.content.push(block);
 640 						stream.push({ type: "text_start", contentIndex: output.content.length - 1, partial: output });
 641 					} else if (event.content_block.type === "thinking") {
 642 						const block: Block = {
 643 							type: "thinking",
 644 							thinking: event.content_block.thinking ?? "",
 645 							thinkingSignature: event.content_block.signature ?? "",
 646 							index: event.index,
 647 						};
 648 						output.content.push(block);
 649 						stream.push({ type: "thinking_start", contentIndex: output.content.length - 1, partial: output });
 650 					} else if (event.content_block.type === "redacted_thinking") {
 651 						const block: Block = {
 652 							type: "thinking",
 653 							thinking: "[Reasoning redacted]",
 654 							thinkingSignature: event.content_block.data,
 655 							redacted: true,
 656 							index: event.index,
 657 						};
 658 						output.content.push(block);
 659 						stream.push({ type: "thinking_start", contentIndex: output.content.length - 1, partial: output });
 660 					} else if (event.content_block.type === "tool_use") {
 661 						const block: Block = {
 662 							type: "toolCall",
 663 							id: event.content_block.id,
 664 							name: isOAuth
 665 								? fromClaudeCodeName(event.content_block.name, currentTools)
 666 								: event.content_block.name,
 667 							arguments: (event.content_block.input as Record<string, any>) ?? {},
 668 							partialJson: "",
 669 							index: event.index,
 670 						};
 671 						output.content.push(block);
 672 						stream.push({ type: "toolcall_start", contentIndex: output.content.length - 1, partial: output });
 673 					}
 674 				} else if (event.type === "content_block_delta") {
 675 					if (event.delta.type === "text_delta") {
 676 						const index = blocks.findIndex((b) => b.index === event.index);
 677 						const block = blocks[index];
 678 						if (block && block.type === "text") {
 679 							block.text += event.delta.text;
 680 							stream.push({
 681 								type: "text_delta",
 682 								contentIndex: index,
 683 								delta: event.delta.text,
 684 								partial: output,
 685 							});
 686 						}
 687 					} else if (event.delta.type === "thinking_delta") {
 688 						const index = blocks.findIndex((b) => b.index === event.index);
 689 						const block = blocks[index];
 690 						if (block && block.type === "thinking") {
 691 							block.thinking += event.delta.thinking;
 692 							stream.push({
 693 								type: "thinking_delta",
 694 								contentIndex: index,
 695 								delta: event.delta.thinking,
 696 								partial: output,
 697 							});
 698 						}
 699 					} else if (event.delta.type === "input_json_delta") {
 700 						const index = blocks.findIndex((b) => b.index === event.index);
 701 						const block = blocks[index];
 702 						if (block && block.type === "toolCall") {
 703 							block.partialJson += event.delta.partial_json;
 704 							block.arguments = parseStreamingJson(block.partialJson);
 705 							stream.push({
 706 								type: "toolcall_delta",
 707 								contentIndex: index,
 708 								delta: event.delta.partial_json,
 709 								partial: output,
 710 							});
 711 						}
 712 					} else if (event.delta.type === "signature_delta") {
 713 						const index = blocks.findIndex((b) => b.index === event.index);
 714 						const block = blocks[index];
 715 						if (block && block.type === "thinking") {
 716 							block.thinkingSignature = block.thinkingSignature || "";
 717 							block.thinkingSignature += event.delta.signature;
 718 						}
 719 					}
 720 				} else if (event.type === "content_block_stop") {
 721 					const index = blocks.findIndex((b) => b.index === event.index);
 722 					const block = blocks[index];
 723 					if (block) {
 724 						delete (block as any).index;
 725 						if (block.type === "text") {
 726 							stream.push({
 727 								type: "text_end",
 728 								contentIndex: index,
 729 								content: block.text,
 730 								partial: output,
 731 							});
 732 						} else if (block.type === "thinking") {
 733 							stream.push({
 734 								type: "thinking_end",
 735 								contentIndex: index,
 736 								content: block.thinking,
 737 								partial: output,
 738 							});
 739 						} else if (block.type === "toolCall") {
 740 							block.arguments = parseStreamingJson(block.partialJson);
 741 							// Finalize in-place and strip the scratch buffer so replay only
 742 							// carries parsed arguments.
 743 							delete (block as { partialJson?: string }).partialJson;
 744 							stream.push({
 745 								type: "toolcall_end",
 746 								contentIndex: index,
 747 								toolCall: block,
 748 								partial: output,
 749 							});
 750 						}
 751 					}
 752 				} else if (event.type === "message_delta") {
 753 					const transformations = event.input_transformations;
 754 					if (Array.isArray(transformations)) inputTransformations = transformations;
 755 					if (event.delta.stop_reason) {
 756 						output.rawStopReason = event.delta.stop_reason;
 757 						const stopReasonResult = mapStopReason(event.delta.stop_reason, event.delta.stop_details);
 758 						output.stopReason = stopReasonResult.stopReason;
 759 						if (stopReasonResult.errorMessage) {
 760 							output.errorMessage = stopReasonResult.errorMessage;
 761 						}
 762 					}
 763 					// Only update usage fields if present (not null).
 764 					// Preserves input_tokens from message_start when proxies omit it in message_delta.
 765 					if (event.usage) {
 766 						if (event.usage.input_tokens != null) {
 767 							output.usage.input = event.usage.input_tokens;
 768 						}
 769 						if (event.usage.output_tokens != null) {
 770 							output.usage.output = event.usage.output_tokens;
 771 						}
 772 						if (event.usage.cache_read_input_tokens != null) {
 773 							output.usage.cacheRead = event.usage.cache_read_input_tokens;
 774 						}
 775 						if (event.usage.cache_creation_input_tokens != null) {
 776 							output.usage.cacheWrite = event.usage.cache_creation_input_tokens;
 777 						}
 778 						// Anthropic reports reasoning tokens as a subset of output tokens.
 779 						const thinkingTokens = event.usage.output_tokens_details?.thinking_tokens;
 780 						if (thinkingTokens != null) {
 781 							output.usage.reasoning = thinkingTokens;
 782 						}
 783 					}
 784 					// Anthropic doesn't provide total_tokens, compute from components
 785 					output.usage.totalTokens =
 786 						output.usage.input + output.usage.output + output.usage.cacheRead + output.usage.cacheWrite;
 787 					calculateCost(usageModel, output.usage);
 788 				}
 789 			}
 790 
 791 			if (options?.signal?.aborted) {
 792 				throw new Error("Request was aborted");
 793 			}
 794 
 795 			if (output.stopReason === "pending") {
 796 				throw new Error("Anthropic stream ended without a stop reason");
 797 			}
 798 			if (output.stopReason === "aborted" || output.stopReason === "error") {
 799 				throw new Error(output.errorMessage || "An unknown error occurred");
 800 			}
 801 			if (inputTransformations && inputTransformations.length > 0) {
 802 				appendAssistantMessageDiagnostic(output, {
 803 					type: "anthropic_input_transformations",
 804 					timestamp: Date.now(),
 805 					details: {
 806 						transformations: inputTransformations.map((transformation) => ({
 807 							type: transformation.type ?? undefined,
 808 							path: transformation.path ?? undefined,
 809 							reason: transformation.reason ?? undefined,
 810 						})),
 811 					},
 812 				});
 813 			}
 814 
 815 			stream.push({ type: "done", reason: output.stopReason, message: output });
 816 			stream.end();
 817 		} catch (error) {
 818 			for (const block of output.content) {
 819 				delete (block as { index?: number }).index;
 820 				// partialJson is only a streaming scratch buffer; never persist it.
 821 				delete (block as { partialJson?: string }).partialJson;
 822 			}
 823 			output.stopReason = options?.signal?.aborted ? "aborted" : "error";
 824 			output.errorMessage = error instanceof Error ? error.message : JSON.stringify(error);
 825 			stream.push({ type: "error", reason: output.stopReason, error: output });
 826 			stream.end();
 827 		}
 828 	})();
 829 
 830 	return stream;
 831 };

 833 /**
 834  * Map ThinkingLevel to Anthropic effort levels for adaptive thinking.
 835  * Note: effort "max" is available on all adaptive-thinking Claude models, while native
 836  * "xhigh" is only available on Opus 4.7/4.8, Sonnet 5, and Fable 5.
 837  */
 838 function mapThinkingLevelToEffort(
 839 	model: Model<"anthropic-messages">,
 840 	level: SimpleStreamOptions["reasoning"],
 841 ): AnthropicEffort {
 842 	const mapped = level ? model.thinkingLevelMap?.[level] : undefined;
 843 	if (typeof mapped === "string") return mapped as AnthropicEffort;
 844 
 845 	switch (level) {
 846 		case "minimal":
 847 		case "low":
 848 			return "low";
 849 		case "medium":
 850 			return "medium";
 851 		case "high":
 852 			return "high";
 853 		default:
 854 			return "high";
 855 	}
 856 }

 859 	model: Model<"anthropic-messages">,
 860 	context: TranscriptContext,
 861 	options?: SimpleStreamOptions,
 862 ): AssistantMessageEventStream => {
 863 	assertRequestAuth(model.provider, options?.apiKey, options?.headers);
 864 
 865 	const base = {
 866 		...buildBaseOptions(model, context, options, options?.apiKey),
 867 		toolChoice: options?.toolChoice,
 868 	} satisfies AnthropicOptions;
 869 	if (!options?.reasoning) {
 870 		return stream(model, context, {
 871 			...base,
 872 			thinkingEnabled: false,
 873 		} satisfies AnthropicOptions);
 874 	}
 875 
 876 	// For models with adaptive thinking: use an effort level.
 877 	// For older models: use budget-based thinking.

 879 		const effort = mapThinkingLevelToEffort(model, options.reasoning);
 880 		return stream(model, context, {
 881 			...base,

 899 		...base,

 901 		thinkingEnabled: true,
 902 		thinkingBudgetTokens: Math.min(adjusted.thinkingBudget, Math.max(0, maxTokens - 1024)),
 903 	} satisfies AnthropicOptions);
 904 };

 991 function getBetaFeatures(
 992 	model: Model<"anthropic-messages">,
 993 	context: TranscriptContext,
 994 	isOAuthToken: boolean,
 995 	nativeToolChanges: boolean,
 996 	options?: AnthropicOptions,
 997 ): NonNullable<MessageCreateParamsStreaming["betas"]> {
 998 	let configuredFeatures: string | null | undefined;
 999 	for (const headers of [model.headers, options?.headers]) {
1000 		for (const [name, value] of Object.entries(headers ?? {})) {
1001 			if (name.toLowerCase() === "anthropic-beta") configuredFeatures = value;
1002 		}
1003 	}
1004 	if (configuredFeatures === null) return [];
1005 	if (configuredFeatures !== undefined) {
1006 		return [
1007 			...new Set(
1008 				configuredFeatures
1009 					.split(",")
1010 					.map((feature) => feature.trim())
1011 					.filter((feature) => feature.length > 0),
1012 			),
1013 		];
1014 	}
1015 
1016 	const features: NonNullable<MessageCreateParamsStreaming["betas"]> = [];
1017 	if (isOAuthToken) features.push("claude-code-20250219", "oauth-2025-04-20");
1018 	if (shouldUseFineGrainedToolStreamingBeta(model, context)) features.push(FINE_GRAINED_TOOL_STREAMING_BETA);
1019 	if (
1020 		model.reasoning &&
1021 		options?.thinkingEnabled === true &&
1022 		(options.interleavedThinking ?? true) &&
1023 		model.compat?.forceAdaptiveThinking !== true
1024 	) {
1025 		features.push(INTERLEAVED_THINKING_BETA);
1026 	}
1027 	if (shouldUseServerSideFallbackBeta(model)) features.push(SERVER_SIDE_FALLBACK_BETA);
1028 	if (model.compat?.supportsMidConvoEffort === true) {
1029 		features.push(MID_CONVERSATION_OUTPUT_CONFIG_BETA, THINKING_BINDING_CONTROLS_BETA);
1030 	}
1031 	if (nativeToolChanges) features.push(MID_CONVERSATION_TOOL_CHANGES_BETA);
1032 	return [...new Set(features)];
1033 }

1035 function buildParams(
1036 	model: Model<"anthropic-messages">,
1037 	context: TranscriptContext,
1038 	isOAuthToken: boolean,
1039 	options?: AnthropicOptions,
1040 ): MessageCreateParamsStreaming {
1041 	const { cacheControl } = getCacheControl(model, options?.cacheRetention, options?.env);
1042 	const compat = getAnthropicCompat(model);
1043 	const initialSystemMessage = getInitialSystemMessage(context.messages);
1044 	const initialSystemText = initialSystemMessage ? getSystemMessageText(initialSystemMessage) : "";
1045 	const transformedMessages = transformMessages(context.messages, model, normalizeToolCallId);
1046 	const conversationMessages = initialSystemMessage ? transformedMessages.slice(1) : transformedMessages;
1047 	// Native tool changes reference tools by name, so a redefined name cannot be expressed,
1048 	// and Anthropic rejects a tool list where every tool is deferred, so there must be an
1049 	// initial active tool to anchor the deferred ones. Otherwise the current tool list is sent.
1050 	const initialTools = initialSystemMessage?.toolsAdded ?? [];
1051 	const nativeToolChanges =
1052 		compat.supportsMidConvoSystemMessages &&
1053 		compat.supportsMidConvoToolChanges &&
1054 		initialTools.length > 0 &&
1055 		!hasToolRedefinitions(context.messages);
1056 	const converted = convertMessages(
1057 		conversationMessages,
1058 		isOAuthToken,
1059 		cacheControl,
1060 		compat.allowEmptySignature,
1061 		model.compat?.supportsMidConvoEffort === true ? model.provider : undefined,
1062 		nativeToolChanges,
1063 	);
1064 	const activeEffort = options?.effort ?? "high";
1065 	const betaFeatures = getBetaFeatures(model, context, isOAuthToken, nativeToolChanges, options);
1066 	const params: MessageCreateParamsStreaming = {
1067 		model: model.id,
1068 		messages:
1069 			model.compat?.supportsMidConvoEffort === true
1070 				? insertThinkingLevelMessages(converted, activeEffort)
1071 				: converted.messages,
1072 		max_tokens: options?.maxTokens ?? model.maxTokens,
1073 		stream: true,
1074 		...(betaFeatures.length > 0 ? { betas: betaFeatures } : {}),
1075 	};
1076 
1077 	// For OAuth tokens, we MUST include Claude Code identity
1078 	if (isOAuthToken) {
1079 		params.system = [
1080 			{
1081 				type: "text",
1082 				text: "You are Claude Code, Anthropic's official CLI for Claude.",
1083 				...(cacheControl ? { cache_control: cacheControl } : {}),
1084 			},
1085 		];
1086 		if (initialSystemText) {
1087 			params.system.push({
1088 				type: "text",
1089 				text: sanitizeSurrogates(initialSystemText),
1090 				...(cacheControl ? { cache_control: cacheControl } : {}),
1091 			});
1092 		}
1093 	} else if (initialSystemText) {
1094 		// Add cache control to system prompt for non-OAuth tokens
1095 		params.system = [
1096 			{
1097 				type: "text",
1098 				text: sanitizeSurrogates(initialSystemText),
1099 				...(cacheControl ? { cache_control: cacheControl } : {}),
1100 			},
1101 		];
1102 	}
1103 
1104 	// Temperature is incompatible with extended thinking and unsupported on Claude Opus 4.7+.
1105 	if (
1106 		options?.temperature !== undefined &&
1107 		!options?.thinkingEnabled &&
1108 		model.compat?.supportsMidConvoEffort !== true &&
1109 		compat.supportsTemperature
1110 	) {
1111 		params.temperature = options.temperature;
1112 	}
1113 
1114 	const toolCacheControl = compat.supportsCacheControlOnTools ? cacheControl : undefined;
1115 	if (nativeToolChanges) {
1116 		// Initial tools stay active with the cache breakpoint on the last one. Every later
1117 		// declaration is deferred and only surfaced by its `tool_addition` block; removed
1118 		// tools stay declared and are withdrawn by `tool_removal`. The request-level list
1119 		// therefore only grows, keeping the cached prefix intact across tool changes.
1120 		const initialNames = new Set(initialTools.map((tool) => tool.name));
1121 		const laterTools = getDeclaredTools(context.messages).filter((tool) => !initialNames.has(tool.name));
1122 		params.tools = [
1123 			...convertTools(
1124 				initialTools,
1125 				isOAuthToken,
1126 				compat.supportsEagerToolInputStreaming,
1127 				compat.supportsStrictTools,
1128 				toolCacheControl,
1129 			),
1130 			DEFERRED_TOOL_PLACEHOLDER,
1131 			...convertTools(
1132 				laterTools,
1133 				isOAuthToken,
1134 				compat.supportsEagerToolInputStreaming,
1135 				compat.supportsStrictTools,
1136 			).map((tool) => ({ ...tool, defer_loading: true })),
1137 		];
1138 	} else {
1139 		const tools = getCurrentTools(context.messages);
1140 		if (tools.length > 0) {
1141 			params.tools = convertTools(
1142 				tools,
1143 				isOAuthToken,
1144 				compat.supportsEagerToolInputStreaming,
1145 				compat.supportsStrictTools,
1146 				toolCacheControl,
1147 			);
1148 		}
1149 	}
1150 
1151 	// Managed effort models always use adaptive thinking so prefix mismatches can
1152 	// be dropped instead of surfacing as persistent 400 responses.
1153 	if (model.compat?.supportsMidConvoEffort === true) {
1154 		params.thinking = {
1155 			type: "adaptive",
1156 			display: options?.thinkingDisplay ?? "summarized",
1157 			block_binding: { prefix_mismatch_behavior: "drop_block" },
1158 		};
1159 		params.output_config = { effort: "high" };
1160 	} else if (model.reasoning) {
1161 		if (options?.thinkingEnabled) {
1162 			// Default to "summarized" so Opus 4.7 and Mythos Preview behave like
1163 			// older Claude 4 models (whose API default is also "summarized").
1164 			const display: AnthropicThinkingDisplay = options.thinkingDisplay ?? "summarized";
1165 			if (model.compat?.forceAdaptiveThinking === true) {
1166 				// Adaptive thinking: Claude decides when and how much to think.
1167 				params.thinking = { type: "adaptive", display };
1168 				if (options.effort) {
1169 					params.output_config = { effort: options.effort };
1170 				}
1171 			} else {
1172 				// Budget-based thinking for older models

1226 function convertMessages(
1227 	transformedMessages: Message[],
1228 	isOAuthToken: boolean,
1229 	cacheControl?: CacheControlEphemeral,
1230 	allowEmptySignature = false,
1231 	managedProvider?: string,
1232 	nativeToolChanges = false,
1233 ): ConvertedAnthropicMessages {
1234 	const params: MessageParam[] = [];
1235 	const assistantLevels = new Map<number, AnthropicEffort>();
1236 	// Later system messages are held back and emitted directly before the next assistant
1237 	// message (or at the end of the transcript). Anthropic requires `tool_result` blocks to
1238 	// immediately follow their `tool_use`, so a system message between them is rejected; this
1239 	// also mirrors where the managed-effort system messages are inserted. As a result an
1240 	// update placed before a user message in the transcript lands after it on the wire.
1241 	const pendingSystemMessages: MessageParam[] = [];
1242 	const flushPendingSystemMessages = (): void => {
1243 		params.push(...pendingSystemMessages);
1244 		pendingSystemMessages.length = 0;
1245 	};
1246 
1247 	for (let i = 0; i < transformedMessages.length; i++) {
1248 		const msg = transformedMessages[i];
1249 
1250 		if (msg.role === "system") {
1251 			// Later system messages only reach this point when the model accepts them natively;
1252 			// otherwise the transcript was collapsed into the leading message before conversion.
1253 			const text = renderSystemMessageUpdate(msg);
1254 			const blocks: ContentBlockParam[] = [];
1255 			if (text.length > 0) blocks.push({ type: "text", text: sanitizeSurrogates(text) });
1256 			if (nativeToolChanges) {
1257 				for (const tool of msg.toolsRemoved ?? []) {
1258 					blocks.push({
1259 						type: "tool_removal",
1260 						tool: { type: "tool_reference", name: isOAuthToken ? toClaudeCodeName(tool.name) : tool.name },
1261 					});
1262 				}
1263 				for (const tool of msg.toolsAdded ?? []) {
1264 					blocks.push({
1265 						type: "tool_addition",
1266 						tool: { type: "tool_reference", name: isOAuthToken ? toClaudeCodeName(tool.name) : tool.name },
1267 					});
1268 				}
1269 			}
1270 			if (blocks.length > 0) pendingSystemMessages.push({ role: "system", content: blocks });
1271 		} else if (msg.role === "user") {
1272 			if (typeof msg.content === "string") {
1273 				if (msg.content.trim().length > 0) {
1274 					params.push({
1275 						role: "user",
1276 						content: sanitizeSurrogates(msg.content),
1277 					});
1278 				}
1279 			} else {
1280 				const blocks: ContentBlockParam[] = msg.content.map((item) => {
1281 					if (item.type === "text") {
1282 						return {
1283 							type: "text",
1284 							text: sanitizeSurrogates(item.text),
1285 						};
1286 					} else {
1287 						return {
1288 							type: "image",
1289 							source: {
1290 								type: "base64",
1291 								media_type: item.mimeType as "image/jpeg" | "image/png" | "image/gif" | "image/webp",
1292 								data: item.data,
1293 							},
1294 						};
1295 					}
1296 				});
1297 				const filteredBlocks = blocks.filter((b) => {
1298 					if (b.type === "text") {
1299 						return b.text.trim().length > 0;
1300 					}
1301 					return true;
1302 				});
1303 				if (filteredBlocks.length === 0) continue;
1304 				params.push({
1305 					role: "user",
1306 					content: filteredBlocks,
1307 				});
1308 			}
1309 		} else if (msg.role === "assistant") {
1310 			flushPendingSystemMessages();
1311 			const blocks: ContentBlockParam[] = [];
1312 
1313 			for (const block of msg.content) {
1314 				if (block.type === "text") {
1315 					if (block.text.trim().length === 0) continue;
1316 					blocks.push({
1317 						type: "text",
1318 						text: sanitizeSurrogates(block.text),
1319 					});
1320 				} else if (block.type === "thinking") {
1321 					// Redacted thinking: pass the opaque payload back as redacted_thinking
1322 					if (block.redacted) {
1323 						blocks.push({
1324 							type: "redacted_thinking",
1325 							data: block.thinkingSignature!,
1326 						});
1327 						continue;
1328 					}
1329 					const thinkingSignature = block.thinkingSignature;
1330 					const hasThinkingSignature = !!thinkingSignature && thinkingSignature.trim().length > 0;
1331 					if (block.thinking.trim().length === 0 && !hasThinkingSignature) continue;
1332 					// If thinking signature is missing/empty (e.g., from aborted stream),
1333 					// convert to plain text for Anthropic. Some compatible providers emit
1334 					// and accept empty signatures, so let marked models preserve the block.
1335 					if (!hasThinkingSignature) {
1336 						blocks.push(
1337 							allowEmptySignature
1338 								? {
1339 										type: "thinking",
1340 										thinking: sanitizeSurrogates(block.thinking),
1341 										signature: "",
1342 									}
1343 								: {
1344 										type: "text",
1345 										text: sanitizeSurrogates(block.thinking),
1346 									},
1347 						);
1348 					} else {
1349 						blocks.push({
1350 							type: "thinking",
1351 							thinking: sanitizeSurrogates(block.thinking),
1352 							signature: thinkingSignature,
1353 						});
1354 					}
1355 				} else if (block.type === "toolCall") {
1356 					blocks.push({
1357 						type: "tool_use",
1358 						id: block.id,
1359 						name: isOAuthToken ? toClaudeCodeName(block.name) : block.name,
1360 						input: block.arguments ?? {},
1361 					});
1362 				}
1363 			}
1364 			if (blocks.length === 0) continue;
1365 			const messageIndex = params.length;
1366 			params.push({
1367 				role: "assistant",
1368 				content: blocks,
1369 			});
1370 			if (
1371 				managedProvider !== undefined &&
1372 				msg.api === "anthropic-messages" &&
1373 				msg.provider === managedProvider &&
1374 				isAnthropicEffort(msg.providerThinkingLevel)
1375 			) {
1376 				assistantLevels.set(messageIndex, msg.providerThinkingLevel);
1377 			}
1378 		} else if (msg.role === "toolResult") {
1379 			// Collect all consecutive toolResult messages, needed for z.ai Anthropic endpoint.
1380 			const toolResults: ContentBlockParam[] = [];
1381 			let j = i;
1382 			while (j < transformedMessages.length && transformedMessages[j].role === "toolResult") {
1383 				toolResults.push(convertToolResult(transformedMessages[j] as ToolResultMessage));
1384 				j++;
1385 			}
1386 
1387 			// Skip the messages we've already processed.
1388 			i = j - 1;
1389 
1390 			params.push({
1391 				role: "user",
1392 				content: toolResults,
1393 			});
1394 		}
1395 	}
1396 
1397 	flushPendingSystemMessages();
1398 
1399 	// Add cache_control to the last user or system message to cache conversation history
1400 	if (cacheControl && params.length > 0) {
1401 		const lastMessage = params[params.length - 1];
1402 		if (lastMessage.role === "user" || lastMessage.role === "system") {
1403 			if (Array.isArray(lastMessage.content)) {
1404 				const lastBlock = lastMessage.content[lastMessage.content.length - 1];
1405 				if (
1406 					lastBlock &&
1407 					(lastBlock.type === "text" ||
1408 						lastBlock.type === "image" ||
1409 						lastBlock.type === "tool_result" ||
1410 						lastBlock.type === "tool_addition" ||
1411 						lastBlock.type === "tool_removal")
1412 				) {
1413 					(lastBlock as any).cache_control = cacheControl;
1414 				}
1415 			} else if (typeof lastMessage.content === "string") {
1416 				lastMessage.content = [
1417 					{
1418 						type: "text",
1419 						text: lastMessage.content,
1420 						cache_control: cacheControl,
1421 					},
1422 				] as any;
1423 			}
1424 		}
1425 	}
1426 
1427 	return { messages: params, assistantLevels };
1428 }

1494 function mapStopReason(
1495 	reason: BetaStopReason | string,
1496 	stopDetails?: RefusalStopDetails | null,
1497 ): { stopReason: StopReason; errorMessage?: string } {
1498 	switch (reason) {
1499 		case "end_turn":
1500 			return { stopReason: "stop" };
1501 		case "max_tokens":
1502 			return { stopReason: "length" };
1503 		case "tool_use":
1504 			return { stopReason: "toolUse" };
1505 		case "refusal":
1506 			return {
1507 				stopReason: "error",
1508 				errorMessage: stopDetails?.explanation || `The model refused to complete the request`,
1509 			};

1512 		case "stop_sequence":

1514 		case "sensitive": // Content flagged by safety filters (not yet in SDK types)
