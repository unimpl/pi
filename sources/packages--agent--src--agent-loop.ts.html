 101 export async function runAgentLoop(
 102 	prompts: AgentMessage[],
 103 	context: AgentContext,
 104 	config: AgentLoopConfig,
 105 	emit: AgentEventSink,
 106 	signal: AbortSignal | undefined,
 107 	streamFn: StreamFn,
 108 ): Promise<AgentMessage[]> {
 109 	const initialMessages = declareToolChanges(context, prompts);
 110 	const newMessages: AgentMessage[] = [...initialMessages];
 111 	const currentContext: AgentContext = {
 112 		...context,
 113 		messages: [...context.messages, ...initialMessages],
 114 	};
 115 
 116 	await emit({ type: "agent_start" });
 117 	await emit({ type: "turn_start" });
 118 	for (const message of initialMessages) {
 119 		await emit({ type: "message_start", message });
 120 		await emit({ type: "message_end", message });
 121 	}
 122 
 123 	await runLoop(currentContext, newMessages, config, signal, emit, streamFn ?? getDefaultStreamFn());
 124 	return newMessages;
 125 }

 162 async function runLoop(
 163 	initialContext: AgentContext,
 164 	newMessages: AgentMessage[],
 165 	initialConfig: AgentLoopConfig,
 166 	signal: AbortSignal | undefined,
 167 	emit: AgentEventSink,
 168 	streamFunction: StreamFn,
 169 ): Promise<void> {
 170 	let currentContext = initialContext;
 171 	let config = initialConfig;
 172 	let lastCompletedTurn: PrepareNextTurnContext | undefined;
 173 	// Check for steering messages at start (user may have typed while waiting)
 174 	let pendingMessages: AgentMessage[] = (await config.getSteeringMessages?.()) || [];
 175 
 176 	// Outer loop: continues when queued follow-up messages arrive after agent would stop
 177 	while (true) {
 178 		let hasMoreToolCalls = true;
 179 
 180 		// Inner loop: process tool calls and steering messages
 181 		while (hasMoreToolCalls || pendingMessages.length > 0) {
 182 			let preparedMessages: AgentMessage[] = [];
 183 			if (lastCompletedTurn) {
 184 				const nextTurnSnapshot = await config.prepareNextTurn?.(lastCompletedTurn);
 185 				if (nextTurnSnapshot) {
 186 					currentContext = nextTurnSnapshot.context ?? currentContext;
 187 					preparedMessages = nextTurnSnapshot.messages ?? [];
 188 					config = {
 189 						...config,
 190 						model: nextTurnSnapshot.model ?? config.model,
 191 						reasoning:
 192 							nextTurnSnapshot.thinkingLevel === undefined
 193 								? config.reasoning
 194 								: nextTurnSnapshot.thinkingLevel === "off"
 195 									? undefined
 196 									: nextTurnSnapshot.thinkingLevel,
 197 					};
 198 				}
 199 				// Preparation can be long-running (for example, compaction). Pick up steering
 200 				// queued while it ran. Only poll again if the earlier poll returned nothing;
 201 				// otherwise one-at-a-time mode would deliver two messages in this turn.
 202 				if (pendingMessages.length === 0) {
 203 					pendingMessages = (await config.getSteeringMessages?.()) || [];
 204 				}
 205 				await emit({ type: "turn_start" });
 206 			}
 207 
 208 			// Process prepared and queued messages before the next assistant response.
 209 			for (const message of declareToolChanges(currentContext, [...preparedMessages, ...pendingMessages])) {
 210 				await emit({ type: "message_start", message });
 211 				await emit({ type: "message_end", message });
 212 				currentContext.messages.push(message);
 213 				newMessages.push(message);
 214 			}
 215 			pendingMessages = [];
 216 
 217 			// Stream assistant response
 218 			const message = await streamAssistantResponse(currentContext, config, signal, emit, streamFunction);
 219 			newMessages.push(message);
 220 
 221 			if (message.stopReason === "error" || message.stopReason === "aborted") {
 222 				await emit({ type: "turn_end", message, toolResults: [] });
 223 				await emit({ type: "agent_end", messages: newMessages });
 224 				return;
 225 			}
 226 
 227 			// Check for tool calls
 228 			const toolCalls = message.content.filter((c) => c.type === "toolCall");
 229 
 230 			const toolResults: ToolResultMessage[] = [];
 231 			hasMoreToolCalls = false;
 232 			if (toolCalls.length > 0) {
 233 				// A "length" stop means the output was cut off by the token limit, so
 234 				// every tool call in the message may carry truncated arguments. Fail
 235 				// them all instead of executing potentially borked calls.
 236 				const executedToolBatch =
 237 					message.stopReason === "length"
 238 						? await failToolCallsFromTruncatedMessage(toolCalls, emit)
 239 						: await executeToolCalls(currentContext, message, config, signal, emit);
 240 				toolResults.push(...executedToolBatch.messages);
 241 				hasMoreToolCalls = !executedToolBatch.terminate;
 242 
 243 				for (const result of toolResults) {
 244 					currentContext.messages.push(result);
 245 					newMessages.push(result);
 246 				}
 247 			}
 248 
 249 			await emit({ type: "turn_end", message, toolResults });
 250 
 251 			lastCompletedTurn = {
 252 				message,
 253 				toolResults,
 254 				context: currentContext,
 255 				newMessages,
 256 			};
 257 
 258 			if (await config.shouldStopAfterTurn?.(lastCompletedTurn)) {
 259 				await emit({ type: "agent_end", messages: newMessages });
 260 				return;
 261 			}
 262 
 263 			pendingMessages = (await config.getSteeringMessages?.()) || [];
 264 		}
 265 
 266 		// Agent would stop here. Check for follow-up messages.
 267 		const followUpMessages = (await config.getFollowUpMessages?.()) || [];
 268 		if (followUpMessages.length > 0) {
 269 			// Set as pending so inner loop processes them
 270 			pendingMessages = followUpMessages;
 271 			continue;
 272 		}
 273 
 274 		// No more messages, exit
 275 		break;
 276 	}
 277 
 278 	await emit({ type: "agent_end", messages: newMessages });
 279 }

 281 /**
 282  * Declare tool loadout changes to the model.
 283  *
 284  * `context.tools` is what the runtime can execute; the transcript's system messages declare
 285  * what the model may call. Before each request the difference becomes `toolsAdded` and
 286  * `toolsRemoved` on a system message. When a pending system message exists, its tool fields
 287  * are treated as intent and replaced with the delta between the committed transcript and
 288  * the executable set, so replay always yields exactly `context.tools`. Otherwise a new
 289  * system message is inserted before the first non-system pending message.
 290  */
 291 function declareToolChanges(context: AgentContext, pendingMessages: AgentMessage[]): AgentMessage[] {
 292 	let systemIndex = -1;
 293 	for (let i = pendingMessages.length - 1; i >= 0; i--) {
 294 		if (pendingMessages[i].role === "system") {
 295 			systemIndex = i;
 296 			break;
 297 		}
 298 	}
 299 	const pending = pendingMessages[systemIndex] as SystemMessage | undefined;
 300 	const baseline = pending
 301 		? pendingMessages.map((message, index) =>
 302 				index === systemIndex ? withToolChanges(pending, NO_CHANGES) : message,
 303 			)
 304 		: pendingMessages;
 305 	const changes = getToolStateChanges(
 306 		getCurrentTools([...context.messages, ...baseline]),
 307 		(context.tools ?? []).map(toToolDeclaration),
 308 	);
 309 	const unchanged = changes.toolsAdded.length === 0 && changes.toolsRemoved.length === 0;
 310 
 311 	if (pending) {
 312 		// Keep the caller's message object when it already declares no tool changes.
 313 		if (unchanged && !pending.toolsAdded?.length && !pending.toolsRemoved?.length) return pendingMessages;
 314 		return baseline.map((message, index) => (index === systemIndex ? withToolChanges(pending, changes) : message));
 315 	}
 316 	if (unchanged) return pendingMessages;
 317 	const update = withToolChanges({ role: "system", content: "", timestamp: Date.now() }, changes);
 318 	const insertIndex = pendingMessages.findIndex((message) => message.role !== "system");
 319 	const index = insertIndex === -1 ? pendingMessages.length : insertIndex;
 320 	return [...pendingMessages.slice(0, index), update, ...pendingMessages.slice(index)];
 321 }

 323 const NO_CHANGES: ToolStateChanges = { toolsAdded: [], toolsRemoved: [] };

 325 /** Copy a system message with its tool fields replaced by `changes`; empty lists omit the field. */
 326 function withToolChanges(message: SystemMessage, { toolsAdded, toolsRemoved }: ToolStateChanges): SystemMessage {
 327 	const { toolsAdded: _added, toolsRemoved: _removed, ...rest } = message;
 328 	return {
 329 		...rest,
 330 		...(toolsAdded.length > 0 ? { toolsAdded } : {}),
 331 		...(toolsRemoved.length > 0 ? { toolsRemoved } : {}),
 332 	};
 333 }

 335 /**
 336  * Stream an assistant response from the LLM.
 337  * This is where AgentMessage[] gets transformed to Message[] for the LLM.
 338  */
 339 async function streamAssistantResponse(
 340 	context: AgentContext,
 341 	config: AgentLoopConfig,
 342 	signal: AbortSignal | undefined,
 343 	emit: AgentEventSink,
 344 	streamFunction: StreamFn,
 345 ): Promise<AssistantMessage> {
 346 	// Apply context transform if configured (AgentMessage[] → AgentMessage[])
 347 	let messages = context.messages;
 348 	if (config.transformContext) {
 349 		messages = await config.transformContext(messages, signal);
 350 	}
 351 
 352 	// Convert to LLM-compatible messages (AgentMessage[] → Message[])
 353 	const llmMessages = await config.convertToLlm(messages);
 354 
 355 	const llmContext = normalizeContext({ messages: llmMessages });
 356 
 357 	// Resolve API key (important for expiring tokens)
 358 	const resolvedApiKey =
 359 		(config.getApiKey ? await config.getApiKey(config.model.provider) : undefined) || config.apiKey;
 360 
 361 	const response = await streamFunction(config.model, llmContext, {
 362 		...config,
 363 		apiKey: resolvedApiKey,
 364 		signal,
 365 	});
 366 
 367 	let partialMessage: AssistantMessage | null = null;
 368 	let addedPartial = false;
 369 
 370 	for await (const event of response) {
 371 		switch (event.type) {
 372 			case "start":
 373 				partialMessage = event.partial;
 374 				context.messages.push(partialMessage);
 375 				addedPartial = true;
 376 				await emit({ type: "message_start", message: { ...partialMessage } });
 377 				break;
 378 
 379 			case "text_start":
 380 			case "text_delta":
 381 			case "text_end":
 382 			case "thinking_start":
 383 			case "thinking_delta":
 384 			case "thinking_end":
 385 			case "toolcall_start":
 386 			case "toolcall_delta":
 387 			case "toolcall_end":
 388 				if (partialMessage) {
 389 					partialMessage = event.partial;
 390 					context.messages[context.messages.length - 1] = partialMessage;
 391 					await emit({
 392 						type: "message_update",
 393 						assistantMessageEvent: event,
 394 						message: { ...partialMessage },
 395 					});
 396 				}
 397 				break;
 398 
 399 			case "done":
 400 			case "error": {
 401 				const finalMessage = await response.result();
 402 				if (addedPartial) {
 403 					context.messages[context.messages.length - 1] = finalMessage;
 404 				} else {
 405 					context.messages.push(finalMessage);
 406 				}
 407 				if (!addedPartial) {
 408 					await emit({ type: "message_start", message: { ...finalMessage } });
 409 				}
 410 				await emit({ type: "message_end", message: finalMessage });
 411 				return finalMessage;
 412 			}
 413 		}
 414 	}
 415 
 416 	const finalMessage = await response.result();
 417 	if (addedPartial) {
 418 		context.messages[context.messages.length - 1] = finalMessage;
 419 	} else {
 420 		context.messages.push(finalMessage);
 421 		await emit({ type: "message_start", message: { ...finalMessage } });
 422 	}
 423 	await emit({ type: "message_end", message: finalMessage });
 424 	return finalMessage;
 425 }

 427 /**
 428  * Fail all tool calls from an assistant message that was truncated by the
 429  * output token limit. Streamed tool-call arguments are finalized with a
 430  * best-effort JSON salvage parser, so a truncated message can yield tool calls
 431  * whose arguments parse and validate but are silently incomplete. None of them
 432  * are safe to execute; report each as an error so the model can re-issue them.
 433  */
 434 async function failToolCallsFromTruncatedMessage(
 435 	toolCalls: AgentToolCall[],
 436 	emit: AgentEventSink,
 437 ): Promise<ExecutedToolCallBatch> {
 438 	const messages: ToolResultMessage[] = [];
 439 	for (const toolCall of toolCalls) {
 440 		await emit({
 441 			type: "tool_execution_start",
 442 			toolCallId: toolCall.id,
 443 			toolName: toolCall.name,
 444 			args: toolCall.arguments,
 445 		});
 446 		const finalized: FinalizedToolCallOutcome = {
 447 			toolCall,
 448 			result: createErrorToolResult(
 449 				`Tool call "${toolCall.name}" was not executed: the response hit the output token limit, so its arguments may be truncated. Re-issue the tool call with complete arguments.`,
 450 			),
 451 			isError: true,
 452 		};
 453 		await emitToolExecutionEnd(finalized, emit);
 454 		const toolResultMessage = createToolResultMessage(finalized);
 455 		await emitToolResultMessage(toolResultMessage, emit);
 456 		messages.push(toolResultMessage);
 457 	}
 458 	return { messages, terminate: false };
 459 }

 462  * Execute tool calls from an assistant message.
 463  */
 464 async function executeToolCalls(
 465 	currentContext: AgentContext,
 466 	assistantMessage: AssistantMessage,
 467 	config: AgentLoopConfig,
 468 	signal: AbortSignal | undefined,
 469 	emit: AgentEventSink,
 470 ): Promise<ExecutedToolCallBatch> {
 471 	const toolCalls = assistantMessage.content.filter((c) => c.type === "toolCall");
 472 	const hasSequentialToolCall = toolCalls.some(
 473 		(tc) => currentContext.tools?.find((t) => t.name === tc.name)?.executionMode === "sequential",
 474 	);
 475 	if (config.toolExecution === "sequential" || hasSequentialToolCall) {
 476 		return executeToolCallsSequential(currentContext, assistantMessage, toolCalls, config, signal, emit);
 477 	}
 478 	return executeToolCallsParallel(currentContext, assistantMessage, toolCalls, config, signal, emit);
 479 }

 486 async function executeToolCallsSequential(
 487 	currentContext: AgentContext,
 488 	assistantMessage: AssistantMessage,
 489 	toolCalls: AgentToolCall[],
 490 	config: AgentLoopConfig,
 491 	signal: AbortSignal | undefined,
 492 	emit: AgentEventSink,
 493 ): Promise<ExecutedToolCallBatch> {
 494 	const finalizedCalls: FinalizedToolCallOutcome[] = [];
 495 	const messages: ToolResultMessage[] = [];
 496 
 497 	for (const toolCall of toolCalls) {
 498 		await emit({
 499 			type: "tool_execution_start",
 500 			toolCallId: toolCall.id,
 501 			toolName: toolCall.name,
 502 			args: toolCall.arguments,
 503 		});
 504 
 505 		const preparation = await prepareToolCall(currentContext, assistantMessage, toolCall, config, signal);
 506 		let finalized: FinalizedToolCallOutcome;
 507 		if (preparation.kind === "immediate") {
 508 			finalized = {
 509 				toolCall,
 510 				result: preparation.result,
 511 				isError: preparation.isError,
 512 			};
 513 		} else {
 514 			const executed = await executePreparedToolCall(preparation, signal, emit);
 515 			finalized = await finalizeExecutedToolCall(
 516 				currentContext,
 517 				assistantMessage,
 518 				preparation,
 519 				executed,
 520 				config,
 521 				signal,
 522 			);
 523 		}
 524 
 525 		await emitToolExecutionEnd(finalized, emit);
 526 		const toolResultMessage = createToolResultMessage(finalized);
 527 		await emitToolResultMessage(toolResultMessage, emit);
 528 		finalizedCalls.push(finalized);
 529 		messages.push(toolResultMessage);
 530 
 531 		if (signal?.aborted) {
 532 			break;
 533 		}
 534 	}
 535 
 536 	return {
 537 		messages,
 538 		terminate: shouldTerminateToolBatch(finalizedCalls),
 539 	};
 540 }

 542 async function executeToolCallsParallel(
 543 	currentContext: AgentContext,
 544 	assistantMessage: AssistantMessage,
 545 	toolCalls: AgentToolCall[],
 546 	config: AgentLoopConfig,
 547 	signal: AbortSignal | undefined,
 548 	emit: AgentEventSink,
 549 ): Promise<ExecutedToolCallBatch> {
 550 	const finalizedCalls: FinalizedToolCallEntry[] = [];
 551 
 552 	for (const toolCall of toolCalls) {
 553 		await emit({
 554 			type: "tool_execution_start",
 555 			toolCallId: toolCall.id,
 556 			toolName: toolCall.name,
 557 			args: toolCall.arguments,
 558 		});
 559 
 560 		const preparation = await prepareToolCall(currentContext, assistantMessage, toolCall, config, signal);
 561 		if (preparation.kind === "immediate") {
 562 			const finalized = {
 563 				toolCall,
 564 				result: preparation.result,
 565 				isError: preparation.isError,
 566 			} satisfies FinalizedToolCallOutcome;
 567 			await emitToolExecutionEnd(finalized, emit);
 568 			finalizedCalls.push(finalized);
 569 			if (signal?.aborted) {
 570 				break;
 571 			}
 572 			continue;
 573 		}
 574 
 575 		finalizedCalls.push(async () => {
 576 			if (signal?.aborted) {
 577 				const finalized = {
 578 					toolCall,
 579 					result: createErrorToolResult("Operation aborted"),
 580 					isError: true,
 581 				} satisfies FinalizedToolCallOutcome;
 582 				await emitToolExecutionEnd(finalized, emit);
 583 				return finalized;
 584 			}
 585 			const executed = await executePreparedToolCall(preparation, signal, emit);
 586 			const finalized = await finalizeExecutedToolCall(
 587 				currentContext,
 588 				assistantMessage,
 589 				preparation,
 590 				executed,
 591 				config,
 592 				signal,
 593 			);
 594 			await emitToolExecutionEnd(finalized, emit);
 595 			return finalized;
 596 		});
 597 		if (signal?.aborted) {
 598 			break;
 599 		}
 600 	}
 601 
 602 	const orderedFinalizedCalls = await Promise.all(
 603 		finalizedCalls.map((entry) => (typeof entry === "function" ? entry() : Promise.resolve(entry))),
 604 	);
 605 	const messages: ToolResultMessage[] = [];
 606 	for (const finalized of orderedFinalizedCalls) {
 607 		const toolResultMessage = createToolResultMessage(finalized);
 608 		await emitToolResultMessage(toolResultMessage, emit);
 609 		messages.push(toolResultMessage);
 610 	}
 611 
 612 	return {
 613 		messages,
 614 		terminate: shouldTerminateToolBatch(orderedFinalizedCalls),
 615 	};
 616 }

 648 function prepareToolCallArguments(tool: AgentTool<any>, toolCall: AgentToolCall): AgentToolCall {
 649 	if (!tool.prepareArguments) {

 651 	}

 654 		return toolCall;

 662 async function prepareToolCall(
 663 	currentContext: AgentContext,
 664 	assistantMessage: AssistantMessage,
 665 	toolCall: AgentToolCall,
 666 	config: AgentLoopConfig,
 667 	signal: AbortSignal | undefined,
 668 ): Promise<PreparedToolCall | ImmediateToolCallOutcome> {
 669 	const tool = currentContext.tools?.find((t) => t.name === toolCall.name);
 670 	if (!tool) {
 671 		return {
 672 			kind: "immediate",
 673 			result: createErrorToolResult(`Tool ${toolCall.name} not found`),
 674 			isError: true,
 675 		};
 676 	}
 677 
 678 	try {
 679 		const preparedToolCall = prepareToolCallArguments(tool, toolCall);
 680 		const validatedArgs = validateToolArguments(tool, preparedToolCall);
 681 		if (config.beforeToolCall) {
 682 			const beforeResult = await config.beforeToolCall(
 683 				{
 684 					assistantMessage,
 685 					toolCall,
 686 					args: validatedArgs,
 687 					context: currentContext,
 688 				},
 689 				signal,
 690 			);
 691 			if (signal?.aborted) {
 692 				return {
 693 					kind: "immediate",
 694 					result: createErrorToolResult("Operation aborted"),
 695 					isError: true,
 696 				};
 697 			}
 698 			if (beforeResult?.block) {
 699 				const result = createErrorToolResult(beforeResult.reason || "Tool execution was blocked");
 700 				if (beforeResult.terminate === true) {
 701 					result.terminate = true;
 702 				}
 703 				return {
 704 					kind: "immediate",
 705 					result,
 706 					isError: true,
 707 				};
 708 			}
 709 		}
 710 		if (signal?.aborted) {
 711 			return {
 712 				kind: "immediate",
 713 				result: createErrorToolResult("Operation aborted"),
 714 				isError: true,
 715 			};
 716 		}
 717 		return {
 718 			kind: "prepared",
 719 			toolCall,
 720 			tool,
 721 			args: validatedArgs,
 722 		};
 723 	} catch (error) {
 724 		return {
 725 			kind: "immediate",
 726 			result: createErrorToolResult(error instanceof Error ? error.message : String(error)),
 727 			isError: true,
 728 		};
 729 	}
 730 }

 732 async function executePreparedToolCall(
 733 	prepared: PreparedToolCall,
 734 	signal: AbortSignal | undefined,
 735 	emit: AgentEventSink,
 736 ): Promise<ExecutedToolCallOutcome> {
 737 	const updateEvents: Promise<void>[] = [];
 738 	let acceptingUpdates = true;
 739 
 740 	try {
 741 		const result = await prepared.tool.execute(
 742 			prepared.toolCall.id,
 743 			prepared.args as never,
 744 			signal,
 745 			(partialResult) => {
 746 				if (!acceptingUpdates) return;
 747 				updateEvents.push(
 748 					Promise.resolve(
 749 						emit({
 750 							type: "tool_execution_update",
 751 							toolCallId: prepared.toolCall.id,
 752 							toolName: prepared.toolCall.name,
 753 							args: prepared.toolCall.arguments,
 754 							partialResult,
 755 						}),
 756 					),
 757 				);
 758 			},
 759 		);
 760 		acceptingUpdates = false;
 761 		await Promise.all(updateEvents);
 762 		return { result, isError: false };
 763 	} catch (error) {
 764 		acceptingUpdates = false;
 765 		await Promise.all(updateEvents);
 766 		return {
 767 			result: createErrorToolResult(error instanceof Error ? error.message : String(error)),
 768 			isError: true,
 769 		};
 770 	} finally {
 771 		acceptingUpdates = false;
 772 	}
 773 }

 775 async function finalizeExecutedToolCall(
 776 	currentContext: AgentContext,
 777 	assistantMessage: AssistantMessage,
 778 	prepared: PreparedToolCall,
 779 	executed: ExecutedToolCallOutcome,
 780 	config: AgentLoopConfig,
 781 	signal: AbortSignal | undefined,
 782 ): Promise<FinalizedToolCallOutcome> {
 783 	let result = executed.result;
 784 	let isError = executed.isError;
 785 
 786 	if (config.afterToolCall) {
 787 		try {
 788 			const afterResult = await config.afterToolCall(
 789 				{
 790 					assistantMessage,
 791 					toolCall: prepared.toolCall,
 792 					args: prepared.args,
 793 					result,
 794 					isError,
 795 					context: currentContext,
 796 				},
 797 				signal,
 798 			);
 799 			if (afterResult) {
 800 				result = {
 801 					...result,
 802 					content: afterResult.content ?? result.content,
 803 					details: afterResult.details ?? result.details,
 804 					usage: afterResult.usage ?? result.usage,
 805 					terminate: afterResult.terminate ?? result.terminate,
 806 				};
 807 				isError = afterResult.isError ?? isError;
 808 			}
 809 		} catch (error) {
 810 			result = createErrorToolResult(error instanceof Error ? error.message : String(error));
 811 			isError = true;
 812 		}
 813 	}
 814 
 815 	return {
 816 		toolCall: prepared.toolCall,
 817 		result,
 818 		isError,
 819 	};
 820 }

 822 function createErrorToolResult(message: string): AgentToolResult<any> {
 823 	return {
 824 		content: [{ type: "text", text: message }],
 825 		details: {},
 826 	};

 829 async function emitToolExecutionEnd(finalized: FinalizedToolCallOutcome, emit: AgentEventSink): Promise<void> {

 839 function createToolResultMessage(finalized: FinalizedToolCallOutcome): ToolResultMessage {
 840 	return {
 841 		role: "toolResult",
 842 		toolCallId: finalized.toolCall.id,
 843 		toolName: finalized.toolCall.name,
 844 		// Untyped tools (JS extensions) can return results without content; normalize
 845 		// so the null never enters session history or provider payloads.
 846 		content: finalized.result.content ?? [],
 847 		details: finalized.result.details,
 848 		usage: finalized.result.usage,
 849 		isError: finalized.isError,
 850 		timestamp: Date.now(),
 851 	};
 852 }
