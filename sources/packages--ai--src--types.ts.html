 125 	signal?: AbortSignal;

 127 	telemetryContext?: TelemetryContext;

 129 	/**
 130 	 * Optional fetch implementation for provider HTTP requests.
 131 	 * Defaults to `globalThis.fetch`. Provider adapters that cannot inject a custom implementation may reject it.
 132 	 * This does not affect WebSocket transports.

 143 	 * Return undefined to keep the payload unchanged.
 144 	 */

 146 	/**
 147 	 * Optional callback invoked after an HTTP response is received.
 148 	 */
 149 	onResponse?: (response: ProviderResponse, model: TModel) => void | Promise<void>;
 150 	/**
 151 	 * Optional custom HTTP headers to include in API requests.
 152 	 * Merged with provider defaults; caller values override default headers.
 153 	 * On AWS Bedrock these are injected via a Smithy `build`-step middleware so
 154 	 * they are covered by SigV4 signing; reserved headers (`x-amz-*`,

 156 	 * A null value suppresses a provider/API default header with the same name.
 157 	 */
 158 	headers?: ProviderHeaders;
 159 	/**
 160 	 * HTTP request timeout in milliseconds for providers/SDKs that support it.
 161 	 * For example, OpenAI and Anthropic SDK clients default to 10 minutes.
 162 	 */
 163 	timeoutMs?: number;
 164 	/**
 165 	 * Maximum retry attempts for providers/SDKs that support client-side retries.
 166 	 * For example, OpenAI and Anthropic SDK clients default to 2.
 167 	 */
 168 	maxRetries?: number;
 169 	/**
 170 	 * Maximum delay in milliseconds to wait for a retry when the server requests a long wait.
 171 	 * If the server's requested delay exceeds this value, the request fails immediately

 474 /**
 475  * System instructions and tool declarations at one point in the transcript.
 476  *
 477  * The leading system message is the system prompt. Later system messages change it:
 478  * `content` adds instructions from that point on, `sections` replace or remove named
 479  * prompt sections, and `toolsAdded`/`toolsRemoved` change the tool set. Replaying
 480  * every system message in order yields the current prompt and tools. Providers that
 481  * accept system messages mid-conversation send each one in place; other providers
 482  * rebuild the leading system message from the replayed state.
 483  */
 484 export interface SystemMessage {
 485 	role: "system";
 486 	/** Instruction text. On the leading message this is the base prompt; later, additional instructions. */
 487 	content: string | TextContent[];
 488 	/**
 489 	 * Named, ordered prompt sections rendered verbatim after `content`. The leading message
 490 	 * declares them; later messages replace sections by name, and `null` removes one. Keep
 491 	 * each section self-delimiting (a tag, a heading) so the model can relate an update to
 492 	 * the original. Avoid integer-like names; JSON objects reorder those.
 493 	 */
 494 	sections?: Record<string, string | null>;
 495 	/** Complete definitions of tools that become available at this point. */
 496 	toolsAdded?: Tool[];
 497 	/** Tools that stop being available at this point. */
 498 	toolsRemoved?: ToolReference[];
 499 	timestamp: number; // Unix timestamp in milliseconds
 500 }

 629 /**
 630  * Event protocol for AssistantMessageEventStream.
 631  *
 632  * Successful streams emit `start` before partial updates and terminate with
 633  * `done`. A stream may terminate directly with `error` when request setup fails
 634  * before generation starts; after `start`, failures also terminate with `error`.
 635  * Direct `streamSimple()` calls throw synchronously when request auth is missing.
 636  * Updates and `done` must never appear before `start`.
 637  *
 638  * `partial` is the shared live response-so-far helper, not an event-time
 639  * snapshot. Text and thinking blocks are empty when their `*_start` event is
 640  * emitted and grow only through their corresponding `*_delta` events until the
 641  * authoritative `*_end`. Redacted thinking may be complete at start and emit no
 642  * deltas. Tool-call arguments at `toolcall_start` are provider-specific;
 643  * `toolcall_delta` carries subsequent JSON updates.
 644  */
 645 export type AssistantMessageEvent =
 646 	| { type: "start"; partial: AssistantMessage }
 647 	| { type: "text_start"; contentIndex: number; partial: AssistantMessage }
 648 	| { type: "text_delta"; contentIndex: number; delta: string; partial: AssistantMessage }
 649 	| { type: "text_end"; contentIndex: number; content: string; partial: AssistantMessage }
 650 	| { type: "thinking_start"; contentIndex: number; partial: AssistantMessage }
 651 	| { type: "thinking_delta"; contentIndex: number; delta: string; partial: AssistantMessage }
 652 	| { type: "thinking_end"; contentIndex: number; content: string; partial: AssistantMessage }
 653 	| { type: "toolcall_start"; contentIndex: number; partial: AssistantMessage }
 654 	| { type: "toolcall_delta"; contentIndex: number; delta: string; partial: AssistantMessage }
 655 	| { type: "toolcall_end"; contentIndex: number; toolCall: ToolCall; partial: AssistantMessage }
 656 	| {
 657 			type: "done";
 658 			reason: Extract<StopReason, "stop" | "length" | "toolUse" | "deferred">;
 659 			message: AssistantMessage;
 660 	  }
 661 	| { type: "error"; reason: Extract<StopReason, "aborted" | "error">; error: AssistantMessage };
