 138 /**
 139  * Create an AgentSession with the specified options.
 140  *
 141  * @example
 142  * ```typescript
 143  * // Minimal - uses defaults
 144  * const { session } = await createAgentSession();
 145  *
 146  * // With explicit model
 147  * import { getModel } from '@earendil-works/pi-ai';
 148  * const { session } = await createAgentSession({
 149  *   model: getModel('anthropic', 'claude-opus-4-5'),
 150  *   thinkingLevel: 'high',
 151  * });
 152  *
 153  * // Continue previous session
 154  * const { session, modelFallbackMessage } = await createAgentSession({
 155  *   continueSession: true,
 156  * });
 157  *
 158  * // Full control
 159  * const loader = new DefaultResourceLoader({
 160  *   cwd: process.cwd(),
 161  *   agentDir: getAgentDir(),
 162  *   settingsManager: SettingsManager.create(),
 163  * });
 164  * await loader.reload();
 165  * const { session } = await createAgentSession({
 166  *   model: myModel,
 167  *   tools: ["read", "bash"],
 168  *   resourceLoader: loader,
 169  *   sessionManager: SessionManager.inMemory(),
 170  * });
 171  * ```
 172  */
 173 export async function createAgentSession(options: CreateAgentSessionOptions = {}): Promise<CreateAgentSessionResult> {
 174 	const cwd = resolvePath(options.cwd ?? options.sessionManager?.getCwd() ?? process.cwd());
 175 	const agentDir = options.agentDir ? resolvePath(options.agentDir) : getDefaultAgentDir();
 176 	let resourceLoader = options.resourceLoader;
 177 
 178 	const authPath = options.agentDir ? join(agentDir, "auth.json") : undefined;
 179 	const modelsPath = options.agentDir ? join(agentDir, "models.json") : undefined;
 180 	const modelRuntime = options.modelRuntime ?? (await ModelRuntime.create({ authPath, modelsPath }));
 181 
 182 	const settingsManager = options.settingsManager ?? SettingsManager.create(cwd, agentDir);
 183 	const sessionManager = options.sessionManager ?? SessionManager.create(cwd, getDefaultSessionDir(cwd, agentDir));
 184 
 185 	if (!resourceLoader) {
 186 		resourceLoader = new DefaultResourceLoader({ cwd, agentDir, settingsManager });
 187 		await resourceLoader.reload();
 188 		time("resourceLoader.reload");
 189 	}
 190 
 191 	// Check if session has existing data to restore
 192 	const existingSession = sessionManager.buildSessionContext();
 193 	const hasExistingSession = existingSession.messages.length > 0;
 194 	const hasThinkingEntry = sessionManager.getBranch().some((entry) => entry.type === "thinking_level_change");
 195 
 196 	let model = options.model;
 197 	let modelFallbackMessage: string | undefined;
 198 
 199 	// If session has data, try to restore model from it
 200 	if (!model && hasExistingSession && existingSession.model) {
 201 		const restoredModel = modelRuntime.getModel(existingSession.model.provider, existingSession.model.modelId);
 202 		if (restoredModel && modelRuntime.hasConfiguredAuth(restoredModel.provider)) {
 203 			model = restoredModel;
 204 		}
 205 		if (!model) {
 206 			modelFallbackMessage = `Could not restore model ${existingSession.model.provider}/${existingSession.model.modelId}`;
 207 		}
 208 	}
 209 
 210 	// If still no model, use findInitialModel (checks settings default, then provider defaults)
 211 	if (!model) {
 212 		const result = await findInitialModel({
 213 			scopedModels: [],
 214 			isContinuing: hasExistingSession,
 215 			defaultProvider: settingsManager.getDefaultProvider(),
 216 			defaultModelId: settingsManager.getDefaultModel(),
 217 			defaultThinkingLevel: settingsManager.getDefaultThinkingLevel(),
 218 			modelThinkingLevels: settingsManager.getAllModelThinkingLevels(),
 219 			modelRuntime,
 220 		});
 221 		model = result.model;
 222 		if (!model) {
 223 			modelFallbackMessage = formatNoModelsAvailableMessage();
 224 		} else if (modelFallbackMessage) {
 225 			modelFallbackMessage += `. Using ${model.provider}/${model.id}`;
 226 		}
 227 	}
 228 
 229 	let thinkingLevel = options.thinkingLevel;
 230 
 231 	// If session has data, restore thinking level from it
 232 	if (thinkingLevel === undefined && hasExistingSession) {
 233 		thinkingLevel = hasThinkingEntry
 234 			? (existingSession.thinkingLevel as ThinkingLevel)
 235 			: (settingsManager.getDefaultThinkingLevel() ?? DEFAULT_THINKING_LEVEL);
 236 	}
 237 
 238 	// Fall back to per-model override, then global default
 239 	if (thinkingLevel === undefined && model) {
 240 		const perModel = settingsManager.getModelThinkingLevel(model.provider, model.id);
 241 		if (perModel) {
 242 			thinkingLevel = perModel;
 243 		}
 244 	}
 245 	if (thinkingLevel === undefined) {
 246 		thinkingLevel = settingsManager.getDefaultThinkingLevel() ?? DEFAULT_THINKING_LEVEL;
 247 	}
 248 
 249 	// Clamp to model capabilities
 250 	if (!model) {
 251 		thinkingLevel = "off";
 252 	} else {
 253 		thinkingLevel = clampThinkingLevel(model, thinkingLevel) as ThinkingLevel;
 254 	}
 255 
 256 	const defaultActiveToolNames: ToolName[] = ["read", "bash", "edit", "write"];
 257 	const configuredDefaultToolNames = settingsManager.getDefaultTools();
 258 	const allowedToolNames = options.tools ?? (options.noTools === "all" ? [] : undefined);
 259 	const excludedToolNames = options.excludeTools;
 260 	const excludedToolNameSet = excludedToolNames ? new Set(excludedToolNames) : undefined;
 261 	const initialActiveToolNames = (
 262 		options.tools ?? (options.noTools ? [] : (configuredDefaultToolNames ?? defaultActiveToolNames))
 263 	).filter((name) => !excludedToolNameSet?.has(name));
 264 
 265 	let agent: Agent;
 266 
 267 	// Create convertToLlm wrapper that filters images if blockImages is enabled (defense-in-depth)
 268 	const convertToLlmWithBlockImages = (messages: AgentMessage[]): Message[] => {
 269 		const converted = convertToLlm(messages);
 270 		// Check setting dynamically so mid-session changes take effect
 271 		if (!settingsManager.getBlockImages()) {
 272 			return converted;
 273 		}
 274 		// Filter out ImageContent from all messages, replacing with text placeholder
 275 		return converted.map((msg) => {
 276 			if (msg.role === "user" || msg.role === "toolResult") {
 277 				const content = msg.content;
 278 				if (Array.isArray(content)) {
 279 					const hasImages = content.some((c) => c.type === "image");
 280 					if (hasImages) {
 281 						const filteredContent = content
 282 							.map((c) =>
 283 								c.type === "image" ? { type: "text" as const, text: "Image reading is disabled." } : c,
 284 							)
 285 							.filter(
 286 								(c, i, arr) =>
 287 									// Dedupe consecutive "Image reading is disabled." texts
 288 									!(
 289 										c.type === "text" &&
 290 										c.text === "Image reading is disabled." &&
 291 										i > 0 &&
 292 										arr[i - 1].type === "text" &&
 293 										(arr[i - 1] as { type: "text"; text: string }).text === "Image reading is disabled."
 294 									),
 295 							);
 296 						return { ...msg, content: filteredContent };
 297 					}
 298 				}
 299 			}
 300 			return msg;
 301 		});
 302 	};
 303 
 304 	const extensionRunnerRef: { current?: ExtensionRunner } = {};
 305 
 306 	agent = new Agent({
 307 		initialState: {
 308 			systemPrompt: "",
 309 			model,
 310 			thinkingLevel,
 311 			tools: [],
 312 		},
 313 		convertToLlm: convertToLlmWithBlockImages,
 314 		streamFn: async (model, context, options) => {
 315 			const providerRetrySettings = settingsManager.getProviderRetrySettings();
 316 			const httpIdleTimeoutMs = settingsManager.getHttpIdleTimeoutMs();
 317 			// SDKs treat timeout=0 as 0ms (immediate timeout), not "no timeout".
 318 			// Use max int32 to effectively disable the timeout.
 319 			const effectiveTimeoutMs = httpIdleTimeoutMs === 0 ? 2147483647 : httpIdleTimeoutMs;
 320 			const timeoutMs = options?.timeoutMs ?? providerRetrySettings.timeoutMs ?? effectiveTimeoutMs;
 321 			const websocketConnectTimeoutMs =
 322 				options?.websocketConnectTimeoutMs ?? settingsManager.getWebSocketConnectTimeoutMs();
 323 			const headerRunner = extensionRunnerRef.current;
 324 			return modelRuntime.streamSimple(model, context, {
 325 				...options,
 326 				timeoutMs,
 327 				websocketConnectTimeoutMs,
 328 				maxRetries: options?.maxRetries ?? providerRetrySettings.maxRetries,
 329 				maxRetryDelayMs: options?.maxRetryDelayMs ?? providerRetrySettings.maxRetryDelayMs,
 330 				transformHeaders: async (requestHeaders) => {
 331 					const headers = mergeProviderAttributionHeaders(
 332 						model,
 333 						settingsManager,
 334 						options?.sessionId,
 335 						requestHeaders,
 336 					);
 337 					return headerRunner?.hasHandlers("before_provider_headers")
 338 						? headerRunner.emitBeforeProviderHeaders(headers ?? {})
 339 						: (headers ?? {});
 340 				},
 341 			});
 342 		},
 343 		onPayload: async (payload, _model) => {
 344 			const runner = extensionRunnerRef.current;
 345 			if (!runner?.hasHandlers("before_provider_request")) {
 346 				return payload;
 347 			}
 348 			return runner.emitBeforeProviderRequest(payload);
 349 		},
 350 		onResponse: async (response, _model) => {
 351 			const runner = extensionRunnerRef.current;
 352 			if (!runner?.hasHandlers("after_provider_response")) {
 353 				return;
 354 			}
 355 			await runner.emit({
 356 				type: "after_provider_response",
 357 				status: response.status,
 358 				headers: response.headers,
 359 			});
 360 		},
 361 		sessionId: sessionManager.getSessionId(),
 362 		transformContext: async (messages) => {
 363 			const runner = extensionRunnerRef.current;
 364 			if (!runner) return messages;
 365 			return runner.emitContext(messages);
 366 		},
 367 		steeringMode: settingsManager.getSteeringMode(),
 368 		followUpMode: settingsManager.getFollowUpMode(),
 369 		transport: settingsManager.getTransport(),
 370 		thinkingBudgets: settingsManager.getThinkingBudgets(),
 371 		maxRetryDelayMs: settingsManager.getProviderRetrySettings().maxRetryDelayMs,
 372 	});
 373 
 374 	// Restore messages if session has existing data
 375 	if (hasExistingSession) {
 376 		agent.state.messages = existingSession.messages;
 377 		if (!hasThinkingEntry) {
 378 			sessionManager.appendThinkingLevelChange(thinkingLevel);
 379 		}
 380 	} else {
 381 		// Save initial model and thinking level for new sessions so they can be restored on resume
 382 		if (model) {
 383 			sessionManager.appendModelChange(model.provider, model.id);
 384 		}
 385 		sessionManager.appendThinkingLevelChange(thinkingLevel);
 386 	}
 387 
 388 	const session = new AgentSession({
 389 		agent,
 390 		sessionManager,
 391 		settingsManager,
 392 		cwd,
 393 		scopedModels: options.scopedModels,
 394 		resourceLoader,
 395 		customTools: options.customTools,
 396 		modelRuntime,
 397 		initialActiveToolNames,
 398 		allowedToolNames,
 399 		excludedToolNames,
 400 		extensionRunnerRef,
 401 		sessionStartEvent: options.sessionStartEvent,
 402 	});
 403 	const extensionsResult = resourceLoader.getExtensions();
 404 
 405 	return {
 406 		session,
 407 		extensionsResult,
 408 		modelFallbackMessage,
 409 	};
