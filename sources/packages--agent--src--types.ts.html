  39 /**
  40  * Configuration for how tool calls from a single assistant message are executed.
  41  *
  42  * - "sequential": each tool call is prepared, executed, and finalized before the next one starts.
  43  * - "parallel": tool calls are prepared sequentially, then allowed tools execute concurrently.
  44  *   `tool_execution_end` is emitted in tool completion order after each tool is finalized,
  45  *   while tool-result message artifacts are emitted later in assistant source order.
  46  */
  47 export type ToolExecutionMode = "sequential" | "parallel";

  76 /**
  77  * Partial override returned from `afterToolCall`.
  78  *
  79  * Merge semantics are field-by-field:
  80  * - `content`: if provided, replaces the tool result content array in full
  81  * - `details`: if provided, replaces the tool result details value in full
  82  * - `isError`: if provided, replaces the tool result error flag
  83  * - `usage`: if provided, replaces the tool result usage
  84  * - `terminate`: if provided, replaces the early-termination hint
  85  *

  87  * There is no deep merge for `content`, `details`, or `usage`.

  91 	details?: unknown;
  92 	isError?: boolean;

  94 	usage?: Usage;
  95 	/**
  96 	 * Hint that the agent should stop after the current tool batch.
  97 	 * Early termination only happens when every finalized tool result in the batch sets this to true.

  99 	terminate?: boolean;

 159 	/**
 160 	 * Converts AgentMessage[] to LLM-compatible Message[] before each LLM call.
 161 	 *
 162 	 * Each AgentMessage must be converted to a SystemMessage, UserMessage, AssistantMessage, or ToolResultMessage
 163 	 * that the LLM can understand. AgentMessages that cannot be converted (e.g., UI-only notifications,
 164 	 * status messages) should be filtered out.
 165 	 *
 166 	 * Contract: must not throw or reject. Return a safe fallback value instead.
 167 	 * Throwing interrupts the low-level agent loop without producing a normal event sequence.
 168 	 *
 169 	 * @example
 170 	 * ```typescript
 171 	 * convertToLlm: (messages) => messages.flatMap(m => {
 172 	 *   if (m.role === "custom") {
 173 	 *     // Convert custom message to user message
 174 	 *     return [{ role: "user", content: m.content, timestamp: m.timestamp }];
 175 	 *   }
 176 	 *   if (m.role === "notification") {
 177 	 *     // Filter out UI-only messages
 178 	 *     return [];
 179 	 *   }
 180 	 *   // Pass through standard LLM messages
 181 	 *   return [m];
 182 	 * })
 183 	 * ```
 184 	 */
 185 	convertToLlm: (messages: AgentMessage[]) => Message[] | Promise<Message[]>;
 186 
 187 	/**
 188 	 * Optional transform applied to the context before `convertToLlm`.
 189 	 *
 190 	 * Use this for operations that work at the AgentMessage level:
 191 	 * - Context window management (pruning old messages)
 192 	 * - Injecting context from external sources
 193 	 *
 194 	 * Contract: must not throw or reject. Return the original messages or another
 195 	 * safe fallback value instead.
 196 	 *
 197 	 * @example
 198 	 * ```typescript
 199 	 * transformContext: async (messages) => {
 200 	 *   if (estimateTokens(messages) > MAX_TOKENS) {
 201 	 *     return pruneOldMessages(messages);
 202 	 *   }
 203 	 *   return messages;
 204 	 * }
 205 	 * ```
 206 	 */
 207 	transformContext?: (messages: AgentMessage[], signal?: AbortSignal) => Promise<AgentMessage[]>;
 208 
 209 	/**
 210 	 * Resolves an API key dynamically for each LLM call.
 211 	 *
 212 	 * Useful for short-lived OAuth tokens (e.g., GitHub Copilot) that may expire
 213 	 * during long-running tool execution phases.
 214 	 *
 215 	 * Contract: must not throw or reject. Return undefined when no key is available.
 216 	 */
 217 	getApiKey?: (provider: string) => Promise<string | undefined> | string | undefined;
 218 
 219 	/**
 220 	 * Called after each turn fully completes and `turn_end` has been emitted.
 221 	 *
 222 	 * If it returns true, the loop emits `agent_end` and exits before polling steering or follow-up queues,
 223 	 * without starting another LLM call. The current assistant response and any tool executions finish normally.
 224 	 * This callback sees the completed-turn context and runs before `prepareNextTurn`.
 225 	 *
 226 	 * Use this to request a graceful stop after the current turn, e.g. before context gets too full.
 227 	 *
 228 	 * Contract: must not throw or reject. Throwing interrupts the low-level agent loop without producing a normal event sequence.
 229 	 */
 230 	shouldStopAfterTurn?: (context: ShouldStopAfterTurnContext) => boolean | Promise<boolean>;
 231 
 232 	/**
 233 	 * Called after `turn_end` when the loop will continue, immediately before the next turn starts.
 234 	 * Return replacement context/model/thinking state or messages to append to affect that turn.
 235 	 * Return undefined to keep using the current context/config.
 236 	 */
 237 	prepareNextTurn?: (
 238 		context: PrepareNextTurnContext,
 239 	) => AgentLoopTurnUpdate | undefined | Promise<AgentLoopTurnUpdate | undefined>;
 240 
 241 	/**
 242 	 * Returns steering messages to inject into the conversation mid-run.
 243 	 *
 244 	 * Called after the current assistant turn finishes executing its tool calls, unless `shouldStopAfterTurn` exits first.
 245 	 * If messages are returned, they are added to the context before the next LLM call.
 246 	 * Tool calls from the current assistant message are not skipped.
 247 	 *
 248 	 * Use this for "steering" the agent while it's working.
 249 	 *
 250 	 * Contract: must not throw or reject. Return [] when no steering messages are available.
 251 	 */
 252 	getSteeringMessages?: () => Promise<AgentMessage[]>;
 253 
 254 	/**
 255 	 * Returns follow-up messages to process after the agent would otherwise stop.
 256 	 *
 257 	 * Called when the agent has no more tool calls and no steering messages.
 258 	 * If messages are returned, they're added to the context and the agent
 259 	 * continues with another turn.
 260 	 *
 261 	 * Use this for follow-up messages that should wait until the agent finishes.
 262 	 *
 263 	 * Contract: must not throw or reject. Return [] when no follow-up messages are available.
 264 	 */
 265 	getFollowUpMessages?: () => Promise<AgentMessage[]>;

 267 	/**
 268 	 * Tool execution mode.
 269 	 * - "sequential": execute tool calls one by one
 270 	 * - "parallel": preflight tool calls sequentially, then execute allowed tools concurrently;
 271 	 *   emit `tool_execution_end` in tool completion order after each tool is finalized,
 272 	 *   then emit tool-result message artifacts later in assistant source order
 273 	 *
 274 	 * Default: "parallel"
 275 	 */
 276 	toolExecution?: ToolExecutionMode;
 277 
 278 	/**
 279 	 * Called before a tool is executed, after arguments have been validated.
 280 	 *
 281 	 * Return `{ block: true }` to prevent execution. The loop emits an error tool result instead.
 282 	 * A blocked result can also set `terminate: true` to participate in the batch early-termination rule.
 283 	 * The hook receives the agent abort signal and is responsible for honoring it.
 284 	 */
 285 	beforeToolCall?: (context: BeforeToolCallContext, signal?: AbortSignal) => Promise<BeforeToolCallResult | undefined>;
 286 
 287 	/**

 289 	 *
 290 	 * Return an `AfterToolCallResult` to override parts of the executed tool result:
 291 	 * - `content` replaces the full content array
 292 	 * - `details` replaces the full details payload
 293 	 * - `isError` replaces the error flag
 294 	 * - `usage` replaces the tool result usage
 295 	 * - `terminate` replaces the early-termination hint
 296 	 *
 297 	 * Any omitted fields keep their original values. No deep merge is performed.
 298 	 * The hook receives the agent abort signal and is responsible for honoring it.
 299 	 */

 405 /** Tool definition used by the agent runtime. */
 406 export interface AgentTool<TParameters extends TSchema = TSchema, TDetails = any> extends Tool<TParameters> {
 407 	/** Human-readable label for UI display. */
 408 	label: string;
 409 	/**
 410 	 * Optional compatibility shim for raw tool-call arguments before schema validation.
 411 	 * Must return an object that matches `TParameters`.
 412 	 */
 413 	prepareArguments?: (args: unknown) => Static<TParameters>;
 414 	/** Execute the tool call. Throw on failure instead of encoding errors in `content`. */
 415 	execute: (
 416 		toolCallId: string,
 417 		params: Static<TParameters>,
 418 		signal?: AbortSignal,
 419 		onUpdate?: AgentToolUpdateCallback<TDetails>,
 420 	) => Promise<AgentToolResult<TDetails>>;
 421 	/** Recovery policy for an effect whose durable intent exists but whose outcome is unknown. */
 422 	replay?: "never" | "safe";
 423 	/**
 424 	 * Per-tool execution mode override.
 425 	 * - "sequential": this tool must execute one at a time with other tool calls.
 426 	 * - "parallel": this tool can execute concurrently with other tool calls.
 427 	 *
 428 	 * If omitted, the default execution mode applies.
 429 	 */
 430 	executionMode?: ToolExecutionMode;
 431 }

 441 /**
 442  * Events emitted by the Agent for UI updates.
 443  *
 444  * `agent_end` is the last event emitted for a run, but awaited `Agent.subscribe()`
 445  * listeners for that event are still part of run settlement. The agent becomes
 446  * idle only after those listeners finish.
 447  */
 448 export type AgentEvent =
 449 	// Agent lifecycle
 450 	| { type: "agent_start" }
 451 	| { type: "agent_end"; messages: AgentMessage[] }
 452 	// Turn lifecycle - a turn is one assistant response + any tool calls/results
 453 	| { type: "turn_start" }
 454 	| { type: "turn_end"; message: AgentMessage; toolResults: ToolResultMessage[] }
 455 	// Message lifecycle - emitted for system, user, assistant, and toolResult messages
 456 	| { type: "message_start"; message: AgentMessage }
 457 	// Only emitted for assistant messages during streaming
 458 	| { type: "message_update"; message: AgentMessage; assistantMessageEvent: AssistantMessageEvent }
 459 	| { type: "message_end"; message: AgentMessage }
 460 	// Tool execution lifecycle
 461 	| { type: "tool_execution_start"; toolCallId: string; toolName: string; args: any }
 462 	| { type: "tool_execution_update"; toolCallId: string; toolName: string; args: any; partialResult: any }
 463 	| { type: "tool_execution_end"; toolCallId: string; toolName: string; result: any; isError: boolean };
