   4 	private incoming: T[] = [];
   5 	private outgoing: T[] = [];
   6 
   7 	get length(): number {
   8 		return this.incoming.length + this.outgoing.length;

  11 	enqueue(value: T): void {

  13 	}

  19 			}

  26 export class EventStream<T, R = T> implements AsyncIterable<T> {
  27 	private queue = new FifoQueue<T>();
  28 	private waiting = new FifoQueue<(value: IteratorResult<T>) => void>();
  29 	private done = false;
  30 	private finalResultPromise: Promise<R>;
  31 	private resolveFinalResult!: (result: R) => void;
  32 	private isComplete: (event: T) => boolean;
  33 	private extractResult: (event: T) => R;
  34 
  35 	constructor(isComplete: (event: T) => boolean, extractResult: (event: T) => R) {
  36 		this.isComplete = isComplete;
  37 		this.extractResult = extractResult;
  38 		this.finalResultPromise = new Promise((resolve) => {
  39 			this.resolveFinalResult = resolve;
  40 		});
  41 	}
  42 
  43 	push(event: T): void {
  44 		if (this.done) return;
  45 
  46 		if (this.isComplete(event)) {
  47 			this.done = true;
  48 			this.resolveFinalResult(this.extractResult(event));
  49 		}
  50 
  51 		// Deliver to waiting consumer or queue it
  52 		const waiter = this.waiting.dequeue();
  53 		if (waiter) {
  54 			waiter({ value: event, done: false });
  55 		} else {
  56 			this.queue.enqueue(event);
  57 		}
  58 	}
  59 
  60 	end(result?: R): void {
  61 		this.done = true;
  62 		if (result !== undefined) {
  63 			this.resolveFinalResult(result);
  64 		}
  65 		// Notify all waiting consumers that we're done
  66 		while (this.waiting.length > 0) {
  67 			const waiter = this.waiting.dequeue()!;
  68 			waiter({ value: undefined as any, done: true });
  69 		}
  70 	}
  71 
  72 	async *[Symbol.asyncIterator](): AsyncIterator<T> {
  73 		while (true) {
  74 			if (this.queue.length > 0) {
  75 				yield this.queue.dequeue()!;
  76 			} else if (this.done) {
  77 				return;
  78 			} else {
  79 				const result = await new Promise<IteratorResult<T>>((resolve) => this.waiting.enqueue(resolve));
  80 				if (result.done) return;
  81 				yield result.value;
  82 			}
  83 		}
  84 	}
  85 
  86 	result(): Promise<R> {
  87 		return this.finalResultPromise;
  88 	}
  89 }
